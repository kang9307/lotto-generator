실전 시작 가이드

# AI를 활용한 콘텐츠 제작 워크플로우
from anthropic import Anthropic
from openai import OpenAI
import requests

class AIContentCreator:
    def __init__(self):
        self.claude = Anthropic()
        self.gpt = OpenAI()

    def create_youtube_script(self, topic, duration=60):
        """유튜브 쇼츠/릴스 스크립트 생성"""
        prompt = f"""
        주제: {topic}
        영상 길이: {duration}초

        다음 형식으로 바이럴 쇼츠 스크립트를 작성하세요:
        1. 첫 3초 훅 (강렬한 시작)
        2. 핵심 내용 (구체적 팁/정보)
        3. 행동 유도 (CTA)

        대상: 2030세대, 짧고 강렬하게
        """

        response = self.claude.messages.create(
            model="claude-4-opus",
            max_tokens=2000,
            messages=[{"role": "user", "content": prompt}]
        )

        return response.content[0].text

    def generate_thumbnail_prompt(self, video_topic):
        """썸네일 이미지 프롬프트 생성"""
        prompt = f"Create a viral YouTube thumbnail for: {video_topic}"

        # DALL-E 4로 썸네일 생성
        image_response = self.gpt.images.generate(
            model="dall-e-4",
            prompt=prompt,
            size="1920x1080",
            quality="hd"
        )

        return image_response.data[0].url

    def create_blog_post(self, keyword, word_count=2000):
        """SEO 최적화 블로그 포스트 생성"""
        prompt = f"""
        키워드: {keyword}
        목표 단어 수: {word_count}

        다음을 포함한 블로그 글을 작성하세요:
        1. 매력적인 제목 (CTR 최적화)
        2. 메타 디스크립션
        3. H2, H3 구조화된 본문
        4. 내부 링크 제안
        5. FAQ 섹션
        """

        response = self.gpt.chat.completions.create(
            model="gpt-4.5-turbo",
            messages=[{"role": "user", "content": prompt}]
        )

        return response.choices[0].message.content

# 사용 예시
creator = AIContentCreator()
script = creator.create_youtube_script("2026년 부업 트렌드 TOP 3", 60)
print(script)
