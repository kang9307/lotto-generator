PHP 函数与数组

函数

函数是执行特定任务的代码块，用于实现代码的可复用性和模块化。

函数的定义与调用

<?php
// 基本函数定义
function sayHello() {
    echo "你好！";
}

// 函数调用
sayHello();  // 输出: 你好！

// 带参数的函数
function greet($name) {
    echo "你好，$name！";
}

greet("张三");  // 输出: 你好，张三！

// 带默认值的参数
function greetWithTitle($name, $title = "先生") {
    echo "你好，$name$title！";
}

greetWithTitle("张");  // 输出: 你好，张先生！
greetWithTitle("李", "教授");  // 输出: 你好，李教授！

// 带返回值的函数
function add($a, $b) {
    return $a + $b;
}

$result = add(5, 3);  // $result = 8
echo $result;
?>

函数参数

<?php
// 基本参数
function showProfile($name, $age, $city) {
    echo "$name 今年 $age 岁，住在 $city。";
}

showProfile("张三", 25, "北京");

// 带默认值的参数
function setConfig($host, $user = "admin", $port = 3306) {
    echo "主机: $host, 用户: $user, 端口: $port";
}

setConfig("localhost");  // 主机: localhost, 用户: admin, 端口: 3306
setConfig("example.com", "user1");  // 主机: example.com, 用户: user1, 端口: 3306
setConfig("example.org", "user2", 8080);  // 主机: example.org, 用户: user2, 端口: 8080

// 可变长度参数 (PHP 5.6+)
function sum(...$numbers) {
    $total = 0;
    foreach ($numbers as $number) {
        $total += $number;
    }
    return $total;
}

echo sum(1, 2, 3, 4, 5);  // 15

// 数组展开运算符 (PHP 5.6+)
$values = [1, 2, 3];
echo sum(...$values);  // 6
?>

按引用传递

<?php
// 按值传递（默认）
function incrementValue($value) {
    $value++;
    echo "函数内部: $value<br>";
}

$a = 5;
incrementValue($a);  // 函数内部: 6
echo "函数外部: $a<br>";  // 函数外部: 5 （值未改变）

// 按引用传递
function incrementReference(&$value) {
    $value++;
    echo "函数内部: $value<br>";
}

$b = 5;
incrementReference($b);  // 函数内部: 6
echo "函数外部: $b<br>";  // 函数外部: 6 （值已改变）
?>

类型声明 (PHP 7+)

<?php
// 参数类型声明
function multiply(int $a, int $b) {
    return $a * $b;
}

echo multiply(5, 3);  // 15
// echo multiply("5", "3");  // PHP 7 之前: 15, PHP 7 之后: 15 （自动类型转换）
// 在 PHP 7 严格模式下会抛出 TypeError

// 返回类型声明 (PHP 7+)
function divide(float $a, float $b): float {
    return $a / $b;
}

echo divide(10, 3);  // 3.3333333333333

// 联合类型 (PHP 8+)
function processId(int|string $id): void {
    echo "正在处理的 ID: $id";
}

processId(123);  // 正在处理的 ID: 123
processId("ABC123");  // 正在处理的 ID: ABC123

// 可空类型 (PHP 7.1+)
function findUser(?string $username): ?array {
    if ($username === null) {
        return null;
    }
    // 用户搜索逻辑...
    return ["name" => $username, "active" => true];
}

var_dump(findUser(null));  // NULL
var_dump(findUser("admin"));  // ["name" => "admin", "active" => true]
?>

匿名函数与闭包

<?php
// 匿名函数
$greet = function($name) {
    return "你好，$name！";
};

echo $greet("张三");  // 你好，张三！

// 闭包（使用外部变量）
$greeting = "你好";
$greet = function($name) use ($greeting) {
    return "$greeting，$name！";
};

echo $greet("张三");  // 你好，张三！

// 按引用使用外部变量
$counter = 0;
$increment = function() use (&$counter) {
    $counter++;
    return $counter;
};

echo $increment();  // 1
echo $increment();  // 2
echo $counter;      // 2

// 回调函数
$numbers = [1, 2, 3, 4, 5];
$doubled = array_map(function($n) {
    return $n * 2;
}, $numbers);
print_r($doubled);  // [2, 4, 6, 8, 10]
?>

箭头函数 (PHP 7.4+)

<?php
// 传统的匿名函数
$add = function($a, $b) {
    return $a + $b;
};

// 箭头函数
$add = fn($a, $b) => $a + $b;

echo $add(5, 3);  // 8

// 自动捕获外部变量
$multiplier = 2;
$multiply = fn($n) => $n * $multiplier;

echo $multiply(5);  // 10

// 嵌套箭头函数
$addAndMultiply = fn($a, $b) => fn($c) => ($a + $b) * $c;

$step1 = $addAndMultiply(2, 3);  // fn($c) => (2 + 3) * $c
echo $step1(4);  // 20 ((2 + 3) * 4)
?>

递归函数

<?php
// 计算阶乘
function factorial($n) {
    if ($n <= 1) {
        return 1;
    } else {
        return $n * factorial($n - 1);
    }
}

echo factorial(5);  // 120 (5 * 4 * 3 * 2 * 1)

// 斐波那契数列
function fibonacci($n) {
    if ($n <= 1) {
        return $n;
    } else {
        return fibonacci($n - 1) + fibonacci($n - 2);
    }
}

echo fibonacci(10);  // 55

// 目录遍历
function listFiles($dir) {
    $files = [];

    if (is_dir($dir)) {
        $items = scandir($dir);

        foreach ($items as $item) {
            if ($item != "." && $item != "..") {
                $path = $dir . "/" . $item;

                if (is_dir($path)) {
                    $files = array_merge($files, listFiles($path));
                } else {
                    $files[] = $path;
                }
            }
        }
    }

    return $files;
}

// 注意: 过深的递归可能引发栈溢出
?>

变量作用域与静态变量

<?php
$globalVar = "全局变量";

function testScope() {
    $localVar = "局部变量";

    echo $localVar;  // 局部变量
    // echo $globalVar;  // 错误: 未定义的变量

    global $globalVar;  // 访问全局变量
    echo $globalVar;    // 全局变量

    // 静态变量
    static $counter = 0;
    $counter++;
    echo "计数器: $counter";
}

testScope();  // 局部变量, 全局变量, 计数器: 1
testScope();  // 局部变量, 全局变量, 计数器: 2

// 超全局变量在函数内部始终可访问
function showGlobals() {
    echo $_SERVER['PHP_SELF'];  // 超全局变量无需 global 关键字即可访问
}
?>

数组

数组是 PHP 中最有用的复合数据类型之一，可以在一个变量中存储多个值。

数组的创建与访问

<?php
// 创建索引数组
$fruits = array("苹果", "香蕉", "橙子");
$colors = ["红色", "绿色", "蓝色"];  // PHP 5.4+ 简写语法

// 访问数组元素
echo $fruits[0];  // 苹果
echo $colors[1];  // 绿色

// 修改数组元素
$fruits[1] = "猕猴桃";
echo $fruits[1];  // 猕猴桃

// 添加数组元素
$fruits[] = "芒果";  // 添加到数组末尾
print_r($fruits);  // Array ([0] => 苹果 [1] => 猕猴桃 [2] => 橙子 [3] => 芒果)

// 创建关联数组
$person = array(
    "name" => "张三",
    "age" => 25,
    "city" => "北京"
);

$car = [
    "brand" => "现代",
    "model" => "索纳塔",
    "year" => 2023
];

// 访问关联数组
echo $person["name"];  // 张三
echo $car["model"];    // 索纳塔

// 修改关联数组元素
$person["age"] = 26;

// 添加关联数组元素
$person["email"] = "zhangsan@example.com";
?>

数组遍历

<?php
$fruits = ["苹果", "香蕉", "橙子", "猕猴桃"];

// for 循环
for ($i = 0; $i < count($fruits); $i++) {
    echo $fruits[$i] . "<br>";
}

// foreach 循环（只遍历值）
foreach ($fruits as $fruit) {
    echo $fruit . "<br>";
}

// foreach 循环（索引和值）
foreach ($fruits as $index => $fruit) {
    echo "$index: $fruit<br>";
}

// 关联数组遍历
$person = [
    "name" => "张三",
    "age" => 25,
    "city" => "北京"
];

foreach ($person as $key => $value) {
    echo "$key: $value<br>";
}

// 多维数组遍历
$users = [
    ["name" => "张三", "age" => 25],
    ["name" => "李四", "age" => 30],
    ["name" => "王五", "age" => 28]
];

foreach ($users as $user) {
    echo "姓名: " . $user["name"] . ", 年龄: " . $user["age"] . "<br>";
}

// 嵌套 foreach
foreach ($users as $index => $user) {
    echo "用户 " . ($index + 1) . ":<br>";
    foreach ($user as $key => $value) {
        echo "  $key: $value<br>";
    }
}
?>

数组函数

<?php
$numbers = [5, 3, 8, 1, 2];
$fruits = ["苹果", "香蕉", "橙子"];

// 数组信息
echo count($numbers);           // 5 （数组长度）
echo sizeof($numbers);          // 5 （与 count 等价）
var_dump(empty($numbers));      // bool(false) （检查数组是否为空）
var_dump(isset($numbers[1]));   // bool(true) （检查数组元素是否存在）
var_dump(in_array(3, $numbers)); // bool(true) （检查值是否存在于数组中）
echo array_search("香蕉", $fruits); // 1 （查找值对应的索引/键）

// 数组操作
$combined = array_merge($numbers, $fruits);  // 合并数组
$sliced = array_slice($numbers, 1, 3);       // 数组切片 [3, 8, 1]
array_push($fruits, "猕猴桃", "芒果");       // 在末尾添加元素
$lastFruit = array_pop($fruits);            // 移除并返回最后一个元素 ("芒果")
array_unshift($fruits, "西瓜");             // 在开头添加元素
$firstFruit = array_shift($fruits);         // 移除并返回第一个元素 ("西瓜")

// 数组转换
$doubled = array_map(function($n) {          // 对每个元素应用函数
    return $n * 2;
}, $numbers);

$filtered = array_filter($numbers, function($n) {  // 筛选符合条件的元素
    return $n > 3;
});

$sum = array_reduce($numbers, function($carry, $n) {  // 将数组归约为单一值
    return $carry + $n;
}, 0);

// 排序
sort($numbers);                // 按值升序排序（重置索引）
rsort($numbers);               // 按值降序排序
asort($person);                // 按值升序排序（保留键）
arsort($person);               // 按值降序排序（保留键）
ksort($person);                // 按键升序排序
krsort($person);               // 按键降序排序

// 自定义排序
usort($users, function($a, $b) {  // 按值自定义排序
    return $a["age"] <=> $b["age"];  // 按年龄排序
});

// 数组的键和值
$keys = array_keys($person);    // 只提取键
$values = array_values($person); // 只提取值
$flipped = array_flip($person);  // 互换键和值

// 填充数组
$filled = array_fill(0, 5, "值");  // 从索引 0 开始用 "值" 填充 5 个元素
$range = range(1, 10, 2);  // 生成从 1 到 10、步长为 2 的数组 [1, 3, 5, 7, 9]

// 数组交集与差集
$array1 = [1, 2, 3, 4, 5];
$array2 = [3, 4, 5, 6, 7];
$intersection = array_intersect($array1, $array2);  // 交集 [3, 4, 5]
$difference = array_diff($array1, $array2);        // 差集 [1, 2]
?>

数组解构 (PHP 7.1+)

<?php
// 列表解构
$info = ["张三", 25, "北京"];
[$name, $age, $city] = $info;

echo $name;  // 张三
echo $age;   // 25
echo $city;  // 北京

// 仅解构部分元素
[, $age,] = $info;  // 只取第二个元素
echo $age;  // 25

// 带键的数组解构 (PHP 7.1+)
$data = ["name" => "张三", "age" => 25, "city" => "北京"];
["name" => $n, "age" => $a] = $data;

echo $n;  // 张三
echo $a;  // 25

// 嵌套数组解构
$user = [
    "张三",
    ["北京", "朝阳区"],
    ["PHP", "JavaScript", "Python"]
];

[$username, [$city, $district], $languages] = $user;
echo $username;  // 张三
echo $city;      // 北京
echo $district;  // 朝阳区
echo $languages[0];  // PHP
?>

数组展开 (PHP 7.4+)

<?php
$fruits1 = ["苹果", "香蕉"];
$fruits2 = ["橙子", "猕猴桃"];

// PHP 7.4+ 展开运算符
$allFruits = [...$fruits1, ...$fruits2];
// ["苹果", "香蕉", "橙子", "猕猴桃"]

// 在中间添加其他元素
$combined = [...$fruits1, "芒果", ...$fruits2];
// ["苹果", "香蕉", "芒果", "橙子", "猕猴桃"]

// 关联数组展开
$defaults = ["host" => "localhost", "port" => 3306, "user" => "root"];
$custom = ["user" => "admin", "password" => "secret"];

$config = [...$defaults, ...$custom];
// ["host" => "localhost", "port" => 3306, "user" => "admin", "password" => "secret"]
// 重复的键会被后出现的值覆盖
?>
