PHP Object-Oriented Programming

Starting with PHP 5, PHP began supporting object-oriented programming (OOP) in earnest, and related features have been continuously improved in later versions.

Classes and Objects

Defining a Class and Creating Objects

<?php
// Class definition
class User {
    // Properties
    public $name;
    public $email;
    private $password;

    // Constructor
    public function __construct($name, $email, $password) {
        $this->name = $name;
        $this->email = $email;
        $this->password = $password;
    }

    // Methods
    public function login() {
        return "User {$this->email} has logged in.";
    }

    public function getInfo() {
        return "Name: {$this->name}, Email: {$this->email}";
    }

    // Password change method (accesses a private property)
    public function changePassword($newPassword) {
        $this->password = $newPassword;
        return "Password has been changed.";
    }
}

// Create an object
$user1 = new User("John", "john@example.com", "secret123");

// Call object methods
echo $user1->login();  // User john@example.com has logged in.
echo $user1->getInfo();  // Name: John, Email: john@example.com

// Access a public property
echo $user1->name;  // John
$user1->name = "Jane";
echo $user1->name;  // Jane

// Attempt to access a private property
// echo $user1->password;  // Error: cannot access a private property
?>

Access Modifiers

<?php
class Product {
    public $name;        // Accessible from anywhere
    protected $price;    // Accessible only in this class and its subclasses
    private $discount;   // Accessible only within this class

    public function __construct($name, $price, $discount = 0) {
        $this->name = $name;
        $this->price = $price;
        $this->discount = $discount;
    }

    public function getFinalPrice() {
        return $this->price * (1 - $this->discount / 100);
    }

    protected function calculateTax($rate = 10) {
        return $this->getFinalPrice() * ($rate / 100);
    }

    private function applySpecialDiscount() {
        $this->discount += 5;
    }

    public function applyHolidayDiscount() {
        $this->applySpecialDiscount();  // Can call its own private method
        return "Discount applied.";
    }
}

$product = new Product("Smartphone", 1000000, 10);
echo $product->name;  // Smartphone
echo $product->getFinalPrice();  // 900000

// Error: cannot access a protected property
// echo $product->price;

// Error: cannot access a protected method
// $product->calculateTax();

// Error: cannot access a private method
// $product->applySpecialDiscount();

// Indirectly call the private method through a public method
echo $product->applyHolidayDiscount();  // Discount applied.
?>

Inheritance

<?php
// Parent class
class Vehicle {
    protected $brand;
    protected $color;

    public function __construct($brand, $color) {
        $this->brand = $brand;
        $this->color = $color;
    }

    public function getInfo() {
        return "Brand: {$this->brand}, Color: {$this->color}";
    }

    protected function startEngine() {
        return "Engine started!";
    }
}

// Child class
class Car extends Vehicle {
    private $doors;

    public function __construct($brand, $color, $doors) {
        parent::__construct($brand, $color);  // Call the parent constructor
        $this->doors = $doors;
    }

    public function getInfo() {
        // Override and extend the parent method
        return parent::getInfo() . ", Doors: {$this->doors}";
    }

    public function drive() {
        $engineStatus = $this->startEngine();  // Call the inherited protected method
        return $engineStatus . " The car is driving.";
    }
}

$car = new Car("Hyundai", "Black", 4);
echo $car->getInfo();  // Brand: Hyundai, Color: Black, Doors: 4
echo $car->drive();    // Engine started! The car is driving.

// Check the object's type with the instanceof operator
var_dump($car instanceof Car);      // bool(true)
var_dump($car instanceof Vehicle);  // bool(true)
?>

Constants

<?php
class Math {
    // Class constants
    const PI = 3.14159;
    const E = 2.71828;

    public function getCircleArea($radius) {
        return self::PI * $radius * $radius;
    }
}

// Access the constant
echo Math::PI;  // 3.14159

$math = new Math();
echo $math->getCircleArea(5);  // about 78.5397

// PHP 7.1+ supports access modifiers on constants
class Configuration {
    public const VERSION = "1.0.0";
    protected const SECRET_KEY = "abc123";
    private const API_ENDPOINT = "https://api.example.com";

    public function getApiUrl() {
        return self::API_ENDPOINT . "/v" . self::VERSION;
    }
}

echo Configuration::VERSION;  // 1.0.0
// echo Configuration::SECRET_KEY;  // Error: cannot access a protected constant
?>

Static Members

<?php
class Database {
    // Static properties
    private static $connection = null;
    private static $connectionCount = 0;

    // Static methods
    public static function connect($host, $user, $password) {
        if (self::$connection === null) {
            // In practice, the database connection would be established here
            self::$connection = "DB connected to $host as $user";
            self::$connectionCount++;
        }

        return self::$connection;
    }

    public static function getConnectionCount() {
        return self::$connectionCount;
    }

    // Instance method
    public function query($sql) {
        if (self::$connection === null) {
            throw new Exception("You must establish a connection first.");
        }

        return "Query executed: $sql";
    }
}

// Call a static method
$dbConnection = Database::connect("localhost", "root", "password");
echo $dbConnection;  // DB connected to localhost as root
echo Database::getConnectionCount();  // 1

// An instance is required to use the instance method
$db = new Database();
echo $db->query("SELECT * FROM users");  // Query executed: SELECT * FROM users
?>

Abstract Classes

<?php
// Abstract class
abstract class Shape {
    protected $color;

    public function __construct($color) {
        $this->color = $color;
    }

    // Regular method
    public function getColor() {
        return $this->color;
    }

    // Abstract methods (must be implemented by subclasses)
    abstract public function getArea();
    abstract public function getPerimeter();
}

// Concrete class (implements the abstract class)
class Circle extends Shape {
    private $radius;

    public function __construct($color, $radius) {
        parent::__construct($color);
        $this->radius = $radius;
    }

    public function getArea() {
        return pi() * $this->radius * $this->radius;
    }

    public function getPerimeter() {
        return 2 * pi() * $this->radius;
    }
}

class Rectangle extends Shape {
    private $width;
    private $height;

    public function __construct($color, $width, $height) {
        parent::__construct($color);
        $this->width = $width;
        $this->height = $height;
    }

    public function getArea() {
        return $this->width * $this->height;
    }

    public function getPerimeter() {
        return 2 * ($this->width + $this->height);
    }
}

// An abstract class cannot be instantiated
// $shape = new Shape("Red");  // Error

$circle = new Circle("Blue", 5);
echo $circle->getColor();   // Blue
echo $circle->getArea();    // about 78.54
echo $circle->getPerimeter();  // about 31.42

$rectangle = new Rectangle("Green", 4, 6);
echo $rectangle->getColor();  // Green
echo $rectangle->getArea();   // 24
echo $rectangle->getPerimeter();  // 20
?>

Interfaces

<?php
// Interface definition
interface Drawable {
    public function draw();
}

interface Resizable {
    public function resize($scale);
}

// Implementing multiple interfaces
class Square implements Drawable, Resizable {
    private $size;

    public function __construct($size) {
        $this->size = $size;
    }

    public function draw() {
        return "Drawing a square. Size: {$this->size}";
    }

    public function resize($scale) {
        $this->size *= $scale;
        return "Size has been changed to {$this->size}.";
    }
}

// Combining inheritance and interfaces
abstract class UIElement {
    protected $id;

    public function __construct($id) {
        $this->id = $id;
    }

    abstract public function render();
}

class Button extends UIElement implements Drawable, Resizable {
    private $width;
    private $height;

    public function __construct($id, $width, $height) {
        parent::__construct($id);
        $this->width = $width;
        $this->height = $height;
    }

    public function render() {
        return "<button id='{$this->id}'>Button</button>";
    }

    public function draw() {
        return "Drawing a button. Size: {$this->width}x{$this->height}";
    }

    public function resize($scale) {
        $this->width *= $scale;
        $this->height *= $scale;
        return "Button size has been changed to {$this->width}x{$this->height}.";
    }
}

// Using the interfaces
$square = new Square(10);
echo $square->draw();    // Drawing a square. Size: 10
echo $square->resize(2); // Size has been changed to 20.

$button = new Button("submit-btn", 100, 50);
echo $button->render();  // <button id='submit-btn'>Button</button>
echo $button->draw();    // Drawing a button. Size: 100x50
echo $button->resize(1.5); // Button size has been changed to 150x75.

// Using an interface as a type hint
function drawElements(array $elements) {
    foreach ($elements as $element) {
        if ($element instanceof Drawable) {
            echo $element->draw() . "<br>";
        }
    }
}

drawElements([$square, $button]);
?>

Traits (PHP 5.4+)

<?php
// Trait definition
trait Logger {
    private $logCount = 0;

    public function log($message) {
        $this->logCount++;
        return date('Y-m-d H:i:s') . " - $message (log #{$this->logCount})";
    }

    public function getLogCount() {
        return $this->logCount;
    }
}

trait FileSystem {
    public function readFile($path) {
        return "Reading file: $path";
    }

    public function writeFile($path, $content) {
        return "Writing file: $path, content: $content";
    }
}

// Using a trait in a class
class User {
    use Logger;

    private $name;

    public function __construct($name) {
        $this->name = $name;
        $this->log("User {$this->name} created");
    }
}

class FileManager {
    // Using multiple traits
    use Logger, FileSystem;

    public function copyFile($source, $destination) {
        $this->log("Copying file: $source -> $destination");
        $content = $this->readFile($source);
        $this->writeFile($destination, $content);
        return "File has been copied.";
    }
}

$user = new User("John");
echo $user->log("Login attempt");  // 2025-06-05 12:34:56 - Login attempt (log #2)
echo $user->getLogCount();  // 2

$fileManager = new FileManager();
echo $fileManager->log("File manager initialized");  // 2025-06-05 12:34:56 - File manager initialized (log #1)
echo $fileManager->copyFile("/path/to/source.txt", "/path/to/destination.txt");
?>

Method Chaining

<?php
class Query {
    private $table;
    private $where = [];
    private $orderBy;
    private $limit;

    public function from($table) {
        $this->table = $table;
        return $this;  // Return $this to enable method chaining
    }

    public function where($column, $operator, $value) {
        $this->where[] = "$column $operator '$value'";
        return $this;
    }

    public function orderBy($column, $direction = 'ASC') {
        $this->orderBy = "$column $direction";
        return $this;
    }

    public function limit($limit) {
        $this->limit = $limit;
        return $this;
    }

    public function build() {
        $sql = "SELECT * FROM {$this->table}";

        if (!empty($this->where)) {
            $sql .= " WHERE " . implode(" AND ", $this->where);
        }

        if ($this->orderBy) {
            $sql .= " ORDER BY {$this->orderBy}";
        }

        if ($this->limit) {
            $sql .= " LIMIT {$this->limit}";
        }

        return $sql;
    }
}

// Using method chaining
$query = new Query();
$sql = $query->from('users')
             ->where('age', '>', 18)
             ->where('status', '=', 'active')
             ->orderBy('created_at', 'DESC')
             ->limit(10)
             ->build();

echo $sql;
// Output: SELECT * FROM users WHERE age > '18' AND status = 'active' ORDER BY created_at DESC LIMIT 10
?>

Magic Methods

<?php
class Person {
    private $data = [];

    // Magic method: called when accessing a property that has not been set
    public function __get($name) {
        if (array_key_exists($name, $this->data)) {
            return $this->data[$name];
        }
        return null;
    }

    // Magic method: called when assigning a value to a property that has not been set
    public function __set($name, $value) {
        $this->data[$name] = $value;
    }

    // Magic method: called when isset() is invoked
    public function __isset($name) {
        return isset($this->data[$name]);
    }

    // Magic method: called when unset() is invoked
    public function __unset($name) {
        unset($this->data[$name]);
    }

    // Magic method: called when the object is converted to a string
    public function __toString() {
        return "Person object: " . json_encode($this->data);
    }

    // Magic method: called when an undefined method is invoked
    public function __call($name, $arguments) {
        $prefix = substr($name, 0, 3);
        if ($prefix === 'get') {
            $property = lcfirst(substr($name, 3));
            return $this->__get($property);
        } elseif ($prefix === 'set') {
            $property = lcfirst(substr($name, 3));
            $this->__set($property, $arguments[0]);
            return $this;
        }
        throw new Exception("Method '$name' does not exist.");
    }

    // Magic method: called when an undefined static method is invoked
    public static function __callStatic($name, $arguments) {
        return "Static method '$name' was called";
    }

    // Magic method: called when the object is cloned
    public function __clone() {
        // Logic for a deep copy
        $this->data = array_map(function($item) {
            return is_object($item) ? clone $item : $item;
        }, $this->data);
    }

    // Magic method: called when serialize() is invoked
    public function __sleep() {
        // Return the array of properties to serialize
        return array_keys($this->data);
    }

    // Magic method: called when unserialize() is invoked
    public function __wakeup() {
        // Post-deserialization initialization
    }
}

$person = new Person();
$person->name = "John";  // __set() called
$person->age = 25;        // __set() called

echo $person->name;       // __get() called: John
echo isset($person->name); // __isset() called: true

echo $person->getName();   // __call() called: John
$person->setEmail("john@example.com"); // __call() called

echo $person;             // __toString() called: Person object: {"name":"John","age":25,"email":"john@example.com"}

unset($person->age);      // __unset() called

echo Person::sayHello();   // __callStatic() called: Static method 'sayHello' was called
?>
