PHP Functions and Arrays

Functions

A function is a block of code that performs a specific task. Functions are used to promote reusability and modular design.

Defining and Calling Functions

<?php
// Basic function definition
function sayHello() {
    echo "Hello!";
}

// Function call
sayHello();  // Output: Hello!

// Function with a parameter
function greet($name) {
    echo "Hello, $name!";
}

greet("John");  // Output: Hello, John!

// Parameter with a default value
function greetWithTitle($name, $title = "Mr.") {
    echo "Hello, $title $name!";
}

greetWithTitle("John");  // Output: Hello, Mr. John!
greetWithTitle("Smith", "Professor");  // Output: Hello, Professor Smith!

// Function with a return value
function add($a, $b) {
    return $a + $b;
}

$result = add(5, 3);  // $result = 8
echo $result;
?>

Function Parameters

<?php
// Basic parameters
function showProfile($name, $age, $city) {
    echo "$name is $age years old and lives in $city.";
}

showProfile("John", 25, "Seoul");

// Parameters with default values
function setConfig($host, $user = "admin", $port = 3306) {
    echo "Host: $host, User: $user, Port: $port";
}

setConfig("localhost");  // Host: localhost, User: admin, Port: 3306
setConfig("example.com", "user1");  // Host: example.com, User: user1, Port: 3306
setConfig("example.org", "user2", 8080);  // Host: example.org, User: user2, Port: 8080

// Variadic arguments (PHP 5.6+)
function sum(...$numbers) {
    $total = 0;
    foreach ($numbers as $number) {
        $total += $number;
    }
    return $total;
}

echo sum(1, 2, 3, 4, 5);  // 15

// Array spread operator (PHP 5.6+)
$values = [1, 2, 3];
echo sum(...$values);  // 6
?>

Passing by Reference

<?php
// Passing by value (default)
function incrementValue($value) {
    $value++;
    echo "Inside function: $value<br>";
}

$a = 5;
incrementValue($a);  // Inside function: 6
echo "Outside function: $a<br>";  // Outside function: 5 (value unchanged)

// Passing by reference
function incrementReference(&$value) {
    $value++;
    echo "Inside function: $value<br>";
}

$b = 5;
incrementReference($b);  // Inside function: 6
echo "Outside function: $b<br>";  // Outside function: 6 (value changed)
?>

Type Declarations (PHP 7+)

<?php
// Parameter type declarations
function multiply(int $a, int $b) {
    return $a * $b;
}

echo multiply(5, 3);  // 15
// echo multiply("5", "3");  // Before PHP 7: 15, PHP 7+: 15 (automatic type coercion)
// In PHP 7 strict mode this raises a TypeError

// Return type declarations (PHP 7+)
function divide(float $a, float $b): float {
    return $a / $b;
}

echo divide(10, 3);  // 3.3333333333333

// Union types (PHP 8+)
function processId(int|string $id): void {
    echo "Processing ID: $id";
}

processId(123);  // Processing ID: 123
processId("ABC123");  // Processing ID: ABC123

// Nullable types (PHP 7.1+)
function findUser(?string $username): ?array {
    if ($username === null) {
        return null;
    }
    // User lookup logic...
    return ["name" => $username, "active" => true];
}

var_dump(findUser(null));  // NULL
var_dump(findUser("admin"));  // ["name" => "admin", "active" => true]
?>

Variable Functions

<?php
function sayHello() {
    echo "Hello!";
}

function sayGoodbye() {
    echo "Goodbye!";
}

$functionName = "sayHello";
$functionName();  // Hello!

$functionName = "sayGoodbye";
$functionName();  // Goodbye!

// Works with built-in functions too
$arrayFunction = "array_sum";
echo $arrayFunction([1, 2, 3, 4, 5]);  // 15
?>

Anonymous Functions and Closures

<?php
// Anonymous function
$greet = function($name) {
    return "Hello, $name!";
};

echo $greet("John");  // Hello, John!

// Closure (using outer variables)
$greeting = "Hello";
$greet = function($name) use ($greeting) {
    return "$greeting, $name!";
};

echo $greet("John");  // Hello, John!

// Using outer variables by reference
$counter = 0;
$increment = function() use (&$counter) {
    $counter++;
    return $counter;
};

echo $increment();  // 1
echo $increment();  // 2
echo $counter;      // 2

// Callback function
$numbers = [1, 2, 3, 4, 5];
$doubled = array_map(function($n) {
    return $n * 2;
}, $numbers);
print_r($doubled);  // [2, 4, 6, 8, 10]
?>

Arrow Functions (PHP 7.4+)

<?php
// Traditional anonymous function
$add = function($a, $b) {
    return $a + $b;
};

// Arrow function
$add = fn($a, $b) => $a + $b;

echo $add(5, 3);  // 8

// Automatic capture of outer variables
$multiplier = 2;
$multiply = fn($n) => $n * $multiplier;

echo $multiply(5);  // 10

// Nested arrow functions
$addAndMultiply = fn($a, $b) => fn($c) => ($a + $b) * $c;

$step1 = $addAndMultiply(2, 3);  // fn($c) => (2 + 3) * $c
echo $step1(4);  // 20 ((2 + 3) * 4)
?>

Recursive Functions

<?php
// Factorial calculation
function factorial($n) {
    if ($n <= 1) {
        return 1;
    } else {
        return $n * factorial($n - 1);
    }
}

echo factorial(5);  // 120 (5 * 4 * 3 * 2 * 1)

// Fibonacci sequence
function fibonacci($n) {
    if ($n <= 1) {
        return $n;
    } else {
        return fibonacci($n - 1) + fibonacci($n - 2);
    }
}

echo fibonacci(10);  // 55

// Directory traversal
function listFiles($dir) {
    $files = [];

    if (is_dir($dir)) {
        $items = scandir($dir);

        foreach ($items as $item) {
            if ($item != "." && $item != "..") {
                $path = $dir . "/" . $item;

                if (is_dir($path)) {
                    $files = array_merge($files, listFiles($path));
                } else {
                    $files[] = $path;
                }
            }
        }
    }

    return $files;
}

// Note: deep recursion can cause stack overflow
?>

Variable Scope and Static Variables

<?php
$globalVar = "global variable";

function testScope() {
    $localVar = "local variable";

    echo $localVar;  // local variable
    // echo $globalVar;  // Error: undefined variable

    global $globalVar;  // Access the global variable
    echo $globalVar;    // global variable

    // Static variable
    static $counter = 0;
    $counter++;
    echo "Counter: $counter";
}

testScope();  // local variable, global variable, Counter: 1
testScope();  // local variable, global variable, Counter: 2

// Superglobal variables are always accessible inside functions
function showGlobals() {
    echo $_SERVER['PHP_SELF'];  // Superglobals can be accessed without the global keyword
}
?>

Arrays

Arrays are one of the most useful compound data types in PHP, allowing you to store multiple values in a single variable.

Creating and Accessing Arrays

<?php
// Creating an indexed array
$fruits = array("Apple", "Banana", "Orange");
$colors = ["Red", "Green", "Blue"];  // Short syntax (PHP 5.4+)

// Accessing array elements
echo $fruits[0];  // Apple
echo $colors[1];  // Green

// Modifying array elements
$fruits[1] = "Kiwi";
echo $fruits[1];  // Kiwi

// Adding elements to an array
$fruits[] = "Mango";  // Append to the end
print_r($fruits);  // Array ([0] => Apple [1] => Kiwi [2] => Orange [3] => Mango)

// Creating an associative array
$person = array(
    "name" => "John",
    "age" => 25,
    "city" => "Seoul"
);

$car = [
    "brand" => "Hyundai",
    "model" => "Sonata",
    "year" => 2023
];

// Accessing associative array elements
echo $person["name"];  // John
echo $car["model"];    // Sonata

// Modifying associative array elements
$person["age"] = 26;

// Adding associative array elements
$person["email"] = "john@example.com";
?>

Multidimensional Arrays

<?php
// Two-dimensional array
$matrix = [
    [1, 2, 3],
    [4, 5, 6],
    [7, 8, 9]
];

echo $matrix[1][2];  // 6

// Associative multidimensional array
$users = [
    "user1" => [
        "name" => "John",
        "age" => 25,
        "roles" => ["admin", "editor"]
    ],
    "user2" => [
        "name" => "Mike",
        "age" => 30,
        "roles" => ["user"]
    ]
];

echo $users["user1"]["name"];         // John
echo $users["user2"]["roles"][0];     // user

// More complex nested arrays
$company = [
    "name" => "ABC Corp",
    "departments" => [
        "Development" => [
            "employees" => [
                ["name" => "John", "position" => "Team Lead"],
                ["name" => "Mike", "position" => "Developer"]
            ],
            "projects" => ["Website", "Mobile App"]
        ],
        "Marketing" => [
            "employees" => [
                ["name" => "Emily", "position" => "Team Lead"],
                ["name" => "David", "position" => "Designer"]
            ],
            "campaigns" => ["Summer Sale", "Product Launch"]
        ]
    ]
];

echo $company["departments"]["Development"]["employees"][1]["name"];  // Mike
?>

Iterating Over Arrays

<?php
$fruits = ["Apple", "Banana", "Orange", "Kiwi"];

// for loop
for ($i = 0; $i < count($fruits); $i++) {
    echo $fruits[$i] . "<br>";
}

// foreach loop (values only)
foreach ($fruits as $fruit) {
    echo $fruit . "<br>";
}

// foreach loop (index and value)
foreach ($fruits as $index => $fruit) {
    echo "$index: $fruit<br>";
}

// Iterating over an associative array
$person = [
    "name" => "John",
    "age" => 25,
    "city" => "Seoul"
];

foreach ($person as $key => $value) {
    echo "$key: $value<br>";
}

// Iterating over a multidimensional array
$users = [
    ["name" => "John", "age" => 25],
    ["name" => "Mike", "age" => 30],
    ["name" => "Emily", "age" => 28]
];

foreach ($users as $user) {
    echo "Name: " . $user["name"] . ", Age: " . $user["age"] . "<br>";
}

// Nested foreach
foreach ($users as $index => $user) {
    echo "User " . ($index + 1) . ":<br>";
    foreach ($user as $key => $value) {
        echo "  $key: $value<br>";
    }
}
?>

Array Functions

<?php
$numbers = [5, 3, 8, 1, 2];
$fruits = ["Apple", "Banana", "Orange"];

// Array information
echo count($numbers);           // 5 (array length)
echo sizeof($numbers);          // 5 (same as count)
var_dump(empty($numbers));      // bool(false) (check whether the array is empty)
var_dump(isset($numbers[1]));   // bool(true) (check whether an element exists)
var_dump(in_array(3, $numbers)); // bool(true) (check whether a value is in the array)
echo array_search("Banana", $fruits); // 1 (find the index/key of a value)

// Array manipulation
$combined = array_merge($numbers, $fruits);  // Merge arrays
$sliced = array_slice($numbers, 1, 3);       // Partial array [3, 8, 1]
array_push($fruits, "Kiwi", "Mango");        // Append elements to the end
$lastFruit = array_pop($fruits);            // Remove and return the last element ("Mango")
array_unshift($fruits, "Watermelon");       // Prepend an element
$firstFruit = array_shift($fruits);         // Remove and return the first element ("Watermelon")

// Array transformation
$doubled = array_map(function($n) {          // Apply a function to each element
    return $n * 2;
}, $numbers);

$filtered = array_filter($numbers, function($n) {  // Filter elements matching a condition
    return $n > 3;
});

$sum = array_reduce($numbers, function($carry, $n) {  // Reduce an array to a single value
    return $carry + $n;
}, 0);

// Sorting
sort($numbers);                // Sort by value (reindexes)
rsort($numbers);               // Sort by value in reverse order
asort($person);                // Sort by value (preserves keys)
arsort($person);               // Sort by value in reverse order (preserves keys)
ksort($person);                // Sort by key
krsort($person);               // Sort by key in reverse order

// Custom sort
usort($users, function($a, $b) {  // Custom sort by value
    return $a["age"] <=> $b["age"];  // Sort by age
});

// Array keys and values
$keys = array_keys($person);    // Extract keys only
$values = array_values($person); // Extract values only
$flipped = array_flip($person);  // Swap keys and values

// Filling arrays
$filled = array_fill(0, 5, "value");  // Fill 5 elements starting at index 0 with "value"
$range = range(1, 10, 2);  // Array from 1 to 10 incrementing by 2 [1, 3, 5, 7, 9]

// Array intersection and difference
$array1 = [1, 2, 3, 4, 5];
$array2 = [3, 4, 5, 6, 7];
$intersection = array_intersect($array1, $array2);  // Intersection [3, 4, 5]
$difference = array_diff($array1, $array2);        // Difference [1, 2]
?>

Array Destructuring (PHP 7.1+)

<?php
// List destructuring
$info = ["John", 25, "Seoul"];
[$name, $age, $city] = $info;

echo $name;  // John
echo $age;   // 25
echo $city;  // Seoul

// Destructuring only some elements
[, $age,] = $info;  // Grab only the second element
echo $age;  // 25

// Destructuring arrays with keys (PHP 7.1+)
$data = ["name" => "John", "age" => 25, "city" => "Seoul"];
["name" => $n, "age" => $a] = $data;

echo $n;  // John
echo $a;  // 25

// Nested array destructuring
$user = [
    "John",
    ["Seoul", "Gangnam-gu"],
    ["PHP", "JavaScript", "Python"]
];

[$username, [$city, $district], $languages] = $user;
echo $username;  // John
echo $city;      // Seoul
echo $district;  // Gangnam-gu
echo $languages[0];  // PHP
?>

Array Spread (PHP 7.4+)

<?php
$fruits1 = ["Apple", "Banana"];
$fruits2 = ["Orange", "Kiwi"];

// PHP 7.4+ spread operator
$allFruits = [...$fruits1, ...$fruits2];
// ["Apple", "Banana", "Orange", "Kiwi"]

// Adding other elements in the middle
$combined = [...$fruits1, "Mango", ...$fruits2];
// ["Apple", "Banana", "Mango", "Orange", "Kiwi"]

// Spreading associative arrays
$defaults = ["host" => "localhost", "port" => 3306, "user" => "root"];
$custom = ["user" => "admin", "password" => "secret"];

$config = [...$defaults, ...$custom];
// ["host" => "localhost", "port" => 3306, "user" => "admin", "password" => "secret"]
// Duplicate keys are overwritten by the last value
?>
