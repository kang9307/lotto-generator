PHP 运算符与控制结构

运算符

PHP 提供了多种运算符,用于对变量和值进行操作。

算术运算符

<?php
$a = 10;
$b = 3;

echo $a + $b;  // 加法: 13
echo $a - $b;  // 减法: 7
echo $a * $b;  // 乘法: 30
echo $a / $b;  // 除法: 3.3333...
echo $a % $b;  // 取余: 1
echo $a ** $b; // 幂运算: 1000 (PHP 5.6+)
?>

赋值运算符

<?php
$a = 10;      // 基本赋值
$a += 5;      // 等同于 $a = $a + 5(结果: 15)
$a -= 3;      // 等同于 $a = $a - 3(结果: 12)
$a *= 2;      // 等同于 $a = $a * 2(结果: 24)
$a /= 4;      // 等同于 $a = $a / 4(结果: 6)
$a %= 4;      // 等同于 $a = $a % 4(结果: 2)
$a **= 3;     // 等同于 $a = $a ** 3(结果: 8)
?>

字符串运算符

<?php
$a = "你";
$b = "好";

$c = $a . $b;        // 字符串连接: "你好"
$a .= $b;            // 等同于 $a = $a . $b: "你好"
?>

比较运算符

<?php
$a = 10;
$b = 5;
$c = "10";

var_dump($a == $c);   // 宽松比较(仅比较值): bool(true)
var_dump($a === $c);  // 严格比较(同时比较值和类型): bool(false)
var_dump($a != $b);   // 宽松比较(值不相等): bool(true)
var_dump($a !== $c);  // 严格比较(值或类型不同): bool(true)
var_dump($a < $b);    // less than: bool(false)
var_dump($a > $b);    // greater than: bool(true)
var_dump($a <= $b);   // less than or equal to: bool(false)
var_dump($a >= $b);   // greater than or equal to: bool(true)
var_dump($a <=> $b);  // 太空船运算符 (PHP 7+): int(1)
                      // $a 大于 $b 返回 1,相等返回 0,小于返回 -1
?>

递增递减运算符

<?php
$a = 5;

echo $a++;  // 后置递增: 输出 5,之后 $a 变为 6
echo ++$a;  // 前置递增: $a 先递增到 7 再输出 7
echo $a--;  // 后置递减: 输出 7,之后 $a 变为 6
echo --$a;  // 前置递减: $a 先递减到 5 再输出 5
?>

逻辑运算符

<?php
$a = true;
$b = false;

var_dump($a && $b);  // AND: bool(false)
var_dump($a || $b);  // OR: bool(true)
var_dump($a and $b); // AND(低优先级): bool(false)
var_dump($a or $b);  // OR(低优先级): bool(true)
var_dump(!$a);       // NOT: bool(false)
var_dump($a xor $b); // XOR(异或): bool(true)
?>

位运算符

<?php
$a = 5;     // 二进制: 101
$b = 3;     // 二进制: 011

echo $a & $b;   // 按位与: 1 (二进制: 001)
echo $a | $b;   // 按位或: 7 (二进制: 111)
echo $a ^ $b;   // 按位异或: 6 (二进制: 110)
echo ~$a;       // 按位取反: -6 (二进制: ...11010)
echo $a << 1;   // 左移: 10 (二进制: 1010)
echo $a >> 1;   // 右移: 2 (二进制: 10)
?>

数组运算符

<?php
$a = ["name" => "张三", "age" => 25];
$b = ["name" => "李四", "city" => "北京"];

$c = $a + $b;  // 并集: ["name" => "张三", "age" => 25, "city" => "北京"]
                // 键相同时保留第一个数组的值

var_dump($a == $b);   // 是否存在相同键值对: bool(false)
var_dump($a === $b);  // 键值对、顺序、类型是否全部相同: bool(false)
var_dump($a != $b);   // 是否存在不同键值对: bool(true)
var_dump($a <> $b);   // 等同于 $a != $b: bool(true)
var_dump($a !== $b);  // 键值对、顺序、类型是否有任一不同: bool(true)
?>

三元运算符

<?php
$age = 20;
$status = ($age >= 18) ? "成人" : "未成年";  // 成人

// 嵌套三元运算符
$age = 65;
$category = ($age < 18) ? "未成年" : (($age < 65) ? "成人" : "老年人");  // 老年人

// Null 合并运算符 (PHP 7+)
$username = $_GET["user"] ?? "访客";  // $_GET["user"] 为 null 时赋值为 "访客"
?>

类型转换运算符

<?php
$str = "42";
$num = (int)$str;    // 字符串转整数: 42
$float = (float)$str;  // 字符串转浮点数: 42.0
$bool = (bool)$str;    // 字符串转布尔值: true
$array = (array)$str;  // 字符串转数组: ["42"]
$obj = (object)["name" => "张三"];  // 数组转对象
?>

运算符优先级

PHP 运算符具有优先级,可使用括号显式指定运算优先级。

<?php
$result = 5 + 3 * 2;        // 11(先执行乘法)
$result = (5 + 3) * 2;      // 16(先执行括号内的运算)
$result = 5 + 3 * 2 - 4 / 2; // 9 (5 + 6 - 2 = 9)
?>

条件语句

if 语句

<?php
$age = 18;

if ($age >= 18) {
    echo "是成人。";
}

// 当语句只有一行时可省略花括号
if ($age >= 18)
    echo "是成人。";
?>

if-else 语句

<?php
$age = 16;

if ($age >= 18) {
    echo "是成人。";
} else {
    echo "是未成年人。";
}
?>

if-elseif-else 语句

<?php
$score = 85;

if ($score >= 90) {
    echo "A 等级";
} elseif ($score >= 80) {
    echo "B 等级";
} elseif ($score >= 70) {
    echo "C 等级";
} elseif ($score >= 60) {
    echo "D 等级";
} else {
    echo "F 等级";
}
?>

switch 语句

<?php
$day = 3;

switch ($day) {
    case 1:
        echo "星期一";
        break;
    case 2:
        echo "星期二";
        break;
    case 3:
        echo "星期三";
        break;
    case 4:
        echo "星期四";
        break;
    case 5:
        echo "星期五";
        break;
    case 6:
    case 7:
        echo "周末";
        break;
    default:
        echo "无效的日期";
}
?>

match 表达式 (PHP 8.0+)

<?php
$day = 3;

$dayName = match($day) {
    1 => "星期一",
    2 => "星期二",
    3 => "星期三",
    4 => "星期四",
    5 => "星期五",
    6, 7 => "周末",
    default => "无效的日期"
};

echo $dayName;  // 星期三
?>

使用三元运算符表达条件

<?php
$age = 20;
$status = ($age >= 18) ? "成人" : "未成年";
echo $status;  // 成人
?>

Null 合并运算符 (PHP 7+)

<?php
// 等同于 isset($_GET["user"]) ? $_GET["user"] : "访客"
$username = $_GET["user"] ?? "访客";

// 检查多个变量
$name = $firstName ?? $lastName ?? "无名";
?>

Null 安全运算符 (PHP 8+)

<?php
$user = null;
$name = $user?->name;  // $user 为 null 时 $name 也为 null,不会抛出错误

$users = null;
$firstUserName = $users[0]?->name;  // null,不会抛出错误

$callback = null;
$result = $callback?->call();  // null,不会抛出错误
?>

循环语句

for 语句

<?php
for ($i = 0; $i < 5; $i++) {
    echo "编号: $i<br>";
}
?>

while 语句

<?php
$i = 0;
while ($i < 5) {
    echo "编号: $i<br>";
    $i++;
}
?>

do-while 语句

<?php
$i = 0;
do {
    echo "编号: $i<br>";
    $i++;
} while ($i < 5);
?>

foreach 语句

<?php
$colors = ["红", "绿", "蓝"];

// 仅使用值
foreach ($colors as $color) {
    echo "$color<br>";
}

// 同时使用键和值
$person = [
    "name" => "张三",
    "age" => 25,
    "city" => "北京"
];

foreach ($person as $key => $value) {
    echo "$key: $value<br>";
}
?>

break 与 continue

<?php
// break 示例
for ($i = 0; $i < 10; $i++) {
    if ($i == 5) {
        break;  // 终止循环
    }
    echo $i;  // 仅输出 0, 1, 2, 3, 4
}

// continue 示例
for ($i = 0; $i < 10; $i++) {
    if ($i % 2 == 0) {
        continue;  // 跳过当前迭代
    }
    echo $i;  // 仅输出 1, 3, 5, 7, 9
}
?>

嵌套循环

<?php
// 输出九九乘法表
for ($i = 2; $i <= 9; $i++) {
    echo "$i 的乘法表<br>";
    for ($j = 1; $j <= 9; $j++) {
        echo "$i × $j = " . ($i * $j) . "<br>";
    }
    echo "<br>";
}
?>

循环标签与嵌套控制

<?php
outer: for ($i = 0; $i < 5; $i++) {
    for ($j = 0; $j < 5; $j++) {
        if ($i * $j > 10) {
            break outer;  // 终止外层循环
        }
        echo "($i, $j) = " . ($i * $j) . "<br>";
    }
}
?>

其他控制结构

require 与 include

<?php
// 若文件不存在会抛出致命错误(脚本中断)
require 'header.php';

// 若文件不存在仅发出警告(脚本继续执行)
include 'sidebar.php';

// 即使已被包含,仍会再次包含
require 'config.php';
require 'config.php';  // 会执行两次

// 若已被包含则不再重复包含
require_once 'functions.php';
require_once 'functions.php';  // 只执行一次

include_once 'footer.php';
include_once 'footer.php';  // 只执行一次
?>

goto (PHP 5.3+)

<?php
goto a;
echo "这一行不会被执行。";

a:
echo "通过 goto 语句跳转到此处。";

// 在循环中使用示例
for ($i = 0; $i < 10; $i++) {
    if ($i == 5) {
        goto end;
    }
}

end:
echo "已跳出循环。";
?>

闭包与匿名函数

<?php
// 将匿名函数赋值给变量
$greet = function($name) {
    return "你好,$name!";
};

echo $greet("张三");  // 你好,张三!

// 闭包(使用外部变量的匿名函数)
$message = "很高兴见到你";
$greet = function($name) use ($message) {
    return "$message,$name!";
};

echo $greet("张三");  // 很高兴见到你,张三!

// 在数组函数中使用
$numbers = [1, 2, 3, 4, 5];
$doubled = array_map(function($n) {
    return $n * 2;
}, $numbers);  // [2, 4, 6, 8, 10]
?>

箭头函数 (PHP 7.4+)

<?php
// 传统的匿名函数
$double = function($n) {
    return $n * 2;
};

// 箭头函数
$double = fn($n) => $n * 2;

// 箭头函数会自动捕获外部变量
$factor = 3;
$triple = fn($n) => $n * $factor;

// 在数组函数中使用
$numbers = [1, 2, 3, 4, 5];
$squared = array_map(fn($n) => $n ** 2, $numbers);  // [1, 4, 9, 16, 25]
?>
