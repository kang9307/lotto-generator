実践スタートガイド

# AIを活用したコンテンツ制作ワークフロー
from anthropic import Anthropic
from openai import OpenAI
import requests

class AIContentCreator:
    def __init__(self):
        self.claude = Anthropic()
        self.gpt = OpenAI()

    def create_youtube_script(self, topic, duration=60):
        """YouTubeショート/リールのスクリプト生成"""
        prompt = f"""
        テーマ: {topic}
        動画の長さ: {duration}秒

        以下の形式でバイラルショートスクリプトを作成してください：
        1. 最初の3秒のフック（強烈な開始）
        2. 核心コンテンツ（具体的なコツ/情報）
        3. 行動喚起（CTA）

        ターゲット：20～30代、短く強烈に
        """

        response = self.claude.messages.create(
            model="claude-4-opus",
            max_tokens=2000,
            messages=[{"role": "user", "content": prompt}]
        )

        return response.content[0].text

    def generate_thumbnail_prompt(self, video_topic):
        """サムネイル画像プロンプト生成"""
        prompt = f"Create a viral YouTube thumbnail for: {video_topic}"

        # DALL-E 4でサムネイル生成
        image_response = self.gpt.images.generate(
            model="dall-e-4",
            prompt=prompt,
            size="1920x1080",
            quality="hd"
        )

        return image_response.data[0].url

    def create_blog_post(self, keyword, word_count=2000):
        """SEO最適化ブログ記事の生成"""
        prompt = f"""
        キーワード: {keyword}
        目標文字数: {word_count}

        以下を含むブログ記事を作成してください：
        1. 魅力的なタイトル（CTR最適化）
        2. メタディスクリプション
        3. H2、H3構造化された本文
        4. 内部リンク提案
        5. FAQセクション
        """

        response = self.gpt.chat.completions.create(
            model="gpt-4.5-turbo",
            messages=[{"role": "user", "content": prompt}]
        )

        return response.choices[0].message.content

# 使用例
creator = AIContentCreator()
script = creator.create_youtube_script("2026年副業トレンドTOP 3", 60)
print(script)

副業の注意事項

法的考慮事項

  • 雇用契約の確認：会社の規定で副業禁止条項がないかチェック
  • 税務申告：年間300万ウォン超過時は総合所得税の申告が必須
  • 競業禁止：本業と同一業種の副業は法的問題の可能性あり
  • 業務時間の流用禁止：本業の勤務時間中に副業を行うことは絶対禁止

バーンアウト防止

健全な副業生活のための3原則

  1. 睡眠7時間の確保：副業のために睡眠時間を削らない
  2. 週1日の完全休息：本業も副業も休む日を必ず設ける
  3. 運動を週3回：健康管理をおろそかにしない
