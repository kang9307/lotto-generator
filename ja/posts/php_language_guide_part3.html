PHPの演算子と制御構造

演算子

PHPは様々な演算子を提供しており、変数や値を操作することができます。

算術演算子

<?php
$a = 10;
$b = 3;

echo $a + $b;  // 加算: 13
echo $a - $b;  // 減算: 7
echo $a * $b;  // 乗算: 30
echo $a / $b;  // 除算: 3.3333...
echo $a % $b;  // 剰余: 1
echo $a ** $b; // べき乗: 1000 (PHP 5.6+)
?>

代入演算子

<?php
$a = 10;      // 基本代入
$a += 5;      // $a = $a + 5 と同じ (結果: 15)
$a -= 3;      // $a = $a - 3 と同じ (結果: 12)
$a *= 2;      // $a = $a * 2 と同じ (結果: 24)
$a /= 4;      // $a = $a / 4 と同じ (結果: 6)
$a %= 4;      // $a = $a % 4 と同じ (結果: 2)
$a **= 3;     // $a = $a ** 3 と同じ (結果: 8)
?>

文字列演算子

<?php
$a = "こんに";
$b = "ちは";

$c = $a . $b;        // 文字列連結: "こんにちは"
$a .= $b;            // $a = $a . $b と同じ: "こんにちは"
?>

比較演算子

<?php
$a = 10;
$b = 5;
$c = "10";

var_dump($a == $c);   // 緩やかな比較 (値のみ比較): bool(true)
var_dump($a === $c);  // 厳密な比較 (値と型の両方を比較): bool(false)
var_dump($a != $b);   // 緩やかな比較 (値が異なる): bool(true)
var_dump($a !== $c);  // 厳密な比較 (値または型が異なる): bool(true)
var_dump($a < $b);    // less than: bool(false)
var_dump($a > $b);    // greater than: bool(true)
var_dump($a <= $b);   // less than or equal to: bool(false)
var_dump($a >= $b);   // greater than or equal to: bool(true)
var_dump($a <=> $b);  // 宇宙船演算子 (PHP 7+): int(1)
                      // $a が $b より大きければ 1、等しければ 0、小さければ -1 を返す
?>

インクリメント／デクリメント演算子

<?php
$a = 5;

echo $a++;  // 後置インクリメント: 5 を出力、その後 $a は 6
echo ++$a;  // 前置インクリメント: $a を 7 に増やしてから 7 を出力
echo $a--;  // 後置デクリメント: 7 を出力、その後 $a は 6
echo --$a;  // 前置デクリメント: $a を 5 に減らしてから 5 を出力
?>

論理演算子

<?php
$a = true;
$b = false;

var_dump($a && $b);  // AND: bool(false)
var_dump($a || $b);  // OR: bool(true)
var_dump($a and $b); // AND(優先順位が低い): bool(false)
var_dump($a or $b);  // OR(優先順位が低い): bool(true)
var_dump(!$a);       // NOT: bool(false)
var_dump($a xor $b); // XOR(排他的 OR): bool(true)
?>

ビット演算子

<?php
$a = 5;     // 2進数: 101
$b = 3;     // 2進数: 011

echo $a & $b;   // ビット AND: 1 (2進数: 001)
echo $a | $b;   // ビット OR: 7 (2進数: 111)
echo $a ^ $b;   // ビット XOR: 6 (2進数: 110)
echo ~$a;       // ビット NOT: -6 (2進数: ...11010)
echo $a << 1;   // 左シフト: 10 (2進数: 1010)
echo $a >> 1;   // 右シフト: 2 (2進数: 10)
?>

配列演算子

<?php
$a = ["name" => "山田太郎", "age" => 25];
$b = ["name" => "鈴木一郎", "city" => "東京"];

$c = $a + $b;  // 和集合: ["name" => "山田太郎", "age" => 25, "city" => "東京"]
                // 同じキーの場合は最初の配列の値が保持される

var_dump($a == $b);   // 同じキー/値のペアがあるかを確認: bool(false)
var_dump($a === $b);  // キー/値のペア、順序、型が同じかを確認: bool(false)
var_dump($a != $b);   // 異なるキー/値のペアがあるかを確認: bool(true)
var_dump($a <> $b);   // $a != $b と同じ: bool(true)
var_dump($a !== $b);  // キー/値のペア、順序、型のいずれかが異なるかを確認: bool(true)
?>

三項演算子

<?php
$age = 20;
$status = ($age >= 18) ? "成人" : "未成年";  // 成人

// ネストした三項演算子
$age = 65;
$category = ($age < 18) ? "未成年" : (($age < 65) ? "成人" : "高齢者");  // 高齢者

// Null 合体演算子 (PHP 7+)
$username = $_GET["user"] ?? "ゲスト";  // $_GET["user"] が null なら "ゲスト" を代入
?>

キャスト演算子

<?php
$str = "42";
$num = (int)$str;    // 文字列を整数に変換: 42
$float = (float)$str;  // 文字列を浮動小数点数に変換: 42.0
$bool = (bool)$str;    // 文字列を真偽値に変換: true
$array = (array)$str;  // 文字列を配列に変換: ["42"]
$obj = (object)["name" => "山田太郎"];  // 配列をオブジェクトに変換
?>

演算子の優先順位

PHPの演算子には優先順位があり、括弧を用いて明示的に優先順位を指定できます。

<?php
$result = 5 + 3 * 2;        // 11 (乗算が先に実行される)
$result = (5 + 3) * 2;      // 16 (括弧内の演算が先に実行される)
$result = 5 + 3 * 2 - 4 / 2; // 9 (5 + 6 - 2 = 9)
?>

条件文

if 文

<?php
$age = 18;

if ($age >= 18) {
    echo "成人です。";
}

// 文が1行のみの場合は中括弧を省略可能
if ($age >= 18)
    echo "成人です。";
?>

if-else 文

<?php
$age = 16;

if ($age >= 18) {
    echo "成人です。";
} else {
    echo "未成年です。";
}
?>

if-elseif-else 文

<?php
$score = 85;

if ($score >= 90) {
    echo "A 評価";
} elseif ($score >= 80) {
    echo "B 評価";
} elseif ($score >= 70) {
    echo "C 評価";
} elseif ($score >= 60) {
    echo "D 評価";
} else {
    echo "F 評価";
}
?>

switch 文

<?php
$day = 3;

switch ($day) {
    case 1:
        echo "月曜日";
        break;
    case 2:
        echo "火曜日";
        break;
    case 3:
        echo "水曜日";
        break;
    case 4:
        echo "木曜日";
        break;
    case 5:
        echo "金曜日";
        break;
    case 6:
    case 7:
        echo "週末";
        break;
    default:
        echo "不正な日付";
}
?>

match 式 (PHP 8.0+)

<?php
$day = 3;

$dayName = match($day) {
    1 => "月曜日",
    2 => "火曜日",
    3 => "水曜日",
    4 => "木曜日",
    5 => "金曜日",
    6, 7 => "週末",
    default => "不正な日付"
};

echo $dayName;  // 水曜日
?>

三項演算子を使った条件式

<?php
$age = 20;
$status = ($age >= 18) ? "成人" : "未成年";
echo $status;  // 成人
?>

Null 合体演算子 (PHP 7+)

<?php
// isset($_GET["user"]) ? $_GET["user"] : "ゲスト" と同じ
$username = $_GET["user"] ?? "ゲスト";

// 複数変数のチェック
$name = $firstName ?? $lastName ?? "名前なし";
?>

Null 安全演算子 (PHP 8+)

<?php
$user = null;
$name = $user?->name;  // $user が null なら $name も null、エラーは発生しない

$users = null;
$firstUserName = $users[0]?->name;  // null、エラーは発生しない

$callback = null;
$result = $callback?->call();  // null、エラーは発生しない
?>

ループ文

for 文

<?php
for ($i = 0; $i < 5; $i++) {
    echo "番号: $i<br>";
}
?>

while 文

<?php
$i = 0;
while ($i < 5) {
    echo "番号: $i<br>";
    $i++;
}
?>

do-while 文

<?php
$i = 0;
do {
    echo "番号: $i<br>";
    $i++;
} while ($i < 5);
?>

foreach 文

<?php
$colors = ["赤", "緑", "青"];

// 値のみを使用
foreach ($colors as $color) {
    echo "$color<br>";
}

// キーと値の両方を使用
$person = [
    "name" => "山田太郎",
    "age" => 25,
    "city" => "東京"
];

foreach ($person as $key => $value) {
    echo "$key: $value<br>";
}
?>

break と continue

<?php
// break の例
for ($i = 0; $i < 10; $i++) {
    if ($i == 5) {
        break;  // ループを終了
    }
    echo $i;  // 0, 1, 2, 3, 4 のみ出力
}

// continue の例
for ($i = 0; $i < 10; $i++) {
    if ($i % 2 == 0) {
        continue;  // 現在の繰り返しをスキップ
    }
    echo $i;  // 1, 3, 5, 7, 9 のみ出力
}
?>

多重ループ

<?php
// 九九を出力
for ($i = 2; $i <= 9; $i++) {
    echo "{$i}の段<br>";
    for ($j = 1; $j <= 9; $j++) {
        echo "$i × $j = " . ($i * $j) . "<br>";
    }
    echo "<br>";
}
?>

ループのラベルとネスト制御

<?php
outer: for ($i = 0; $i < 5; $i++) {
    for ($j = 0; $j < 5; $j++) {
        if ($i * $j > 10) {
            break outer;  // 外側のループまで抜ける
        }
        echo "($i, $j) = " . ($i * $j) . "<br>";
    }
}
?>

その他の制御構造

require と include

<?php
// ファイルが存在しない場合は致命的エラー (スクリプト停止)
require 'header.php';

// ファイルが存在しない場合は警告のみ (スクリプトは続行)
include 'sidebar.php';

// ファイルがすでに読み込まれていても再度読み込む
require 'config.php';
require 'config.php';  // 2回実行される

// ファイルがすでに読み込まれている場合は再度読み込まない
require_once 'functions.php';
require_once 'functions.php';  // 一度のみ実行

include_once 'footer.php';
include_once 'footer.php';  // 一度のみ実行
?>

goto (PHP 5.3+)

<?php
goto a;
echo "この行は実行されません。";

a:
echo "goto 文を使ってここまでジャンプしました。";

// ループでの使用例
for ($i = 0; $i < 10; $i++) {
    if ($i == 5) {
        goto end;
    }
}

end:
echo "ループを抜けました。";
?>

クロージャと無名関数

<?php
// 無名関数を変数に代入
$greet = function($name) {
    return "こんにちは、$name さん！";
};

echo $greet("山田太郎");  // こんにちは、山田太郎 さん！

// クロージャ（外部変数を利用する無名関数）
$message = "お会いできて嬉しいです";
$greet = function($name) use ($message) {
    return "$message、$name さん！";
};

echo $greet("山田太郎");  // お会いできて嬉しいです、山田太郎 さん！

// 配列関数での使用
$numbers = [1, 2, 3, 4, 5];
$doubled = array_map(function($n) {
    return $n * 2;
}, $numbers);  // [2, 4, 6, 8, 10]
?>

アロー関数 (PHP 7.4+)

<?php
// 従来の無名関数
$double = function($n) {
    return $n * 2;
};

// アロー関数
$double = fn($n) => $n * 2;

// アロー関数は外部変数を自動的にキャプチャする
$factor = 3;
$triple = fn($n) => $n * $factor;

// 配列関数での使用
$numbers = [1, 2, 3, 4, 5];
$squared = array_map(fn($n) => $n ** 2, $numbers);  // [1, 4, 9, 16, 25]
?>
