实战入门指南

# AI辅助内容创作工作流程
from anthropic import Anthropic
from openai import OpenAI
import requests

class AIContentCreator:
    def __init__(self):
        self.claude = Anthropic()
        self.gpt = OpenAI()

    def create_youtube_script(self, topic, duration=60):
        """生成YouTube Shorts/Reels脚本"""
        prompt = f"""
        主题：{topic}
        视频时长：{duration}秒

        请按以下格式编写病毒式短视频脚本：
        1. 前3秒钩子（强烈开场）
        2. 核心内容（具体技巧/信息）
        3. 行动号召（CTA）

        目标受众：20-30岁，简短有力
        """

        response = self.claude.messages.create(
            model="claude-4-opus",
            max_tokens=2000,
            messages=[{"role": "user", "content": prompt}]
        )

        return response.content[0].text

    def generate_thumbnail_prompt(self, video_topic):
        """生成缩略图图像提示"""
        prompt = f"Create a viral YouTube thumbnail for: {video_topic}"

        # 使用DALL-E 4生成缩略图
        image_response = self.gpt.images.generate(
            model="dall-e-4",
            prompt=prompt,
            size="1920x1080",
            quality="hd"
        )

        return image_response.data[0].url

    def create_blog_post(self, keyword, word_count=2000):
        """生成SEO优化的博客文章"""
        prompt = f"""
        关键词：{keyword}
        目标字数：{word_count}

        请撰写包含以下内容的博客文章：
        1. 吸引人的标题（CTR优化）
        2. Meta描述
        3. H2、H3结构化正文
        4. 内链建议
        5. FAQ部分
        """

        response = self.gpt.chat.completions.create(
            model="gpt-4.5-turbo",
            messages=[{"role": "user", "content": prompt}]
        )

        return response.choices[0].message.content

# 使用示例
creator = AIContentCreator()
script = creator.create_youtube_script("2026年副业趋势TOP 3", 60)
print(script)

副业注意事项

法律注意事项

  • 核查劳动合同：确认公司规定中是否有禁止副业的条款
  • 税务申报：年收入超过300万韩元时必须进行综合所得税申报
  • 竞业限制：与主业同行业的副业可能存在法律风险
  • 禁止占用工作时间：绝对不能在主业工作时间从事副业

预防倦怠

健康副业生活的3项原则

  1. 确保7小时睡眠：不因副业而缩短睡眠时间
  2. 每周1天完全休息：必须有主业、副业都不做的休息日
  3. 每周运动3次：不忽视健康管理
