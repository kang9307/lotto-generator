2. LangChain & LangGraph - Building Complex AI Workflows

# Multi-agent collaboration with LangGraph
from langgraph.graph import StateGraph
from langchain_anthropic import ChatAnthropic
from langchain_openai import ChatOpenAI

# Define research agents
research_agent = ChatAnthropic(model="claude-4-opus")
writing_agent = ChatOpenAI(model="gpt-4.5-turbo")
review_agent = ChatAnthropic(model="claude-4-sonnet")

# Create a workflow graph
workflow = StateGraph()

# Add nodes
workflow.add_node("research", lambda x: research_agent.invoke(x))
workflow.add_node("write", lambda x: writing_agent.invoke(x))
workflow.add_node("review", lambda x: review_agent.invoke(x))

# Connect edges (define the workflow)
workflow.add_edge("research", "write")
workflow.add_edge("write", "review")
workflow.add_conditional_edges(
    "review",
    lambda x: "write" if x.score < 8 else "end",
    {"write": "write", "end": END}
)

# Run
app = workflow.compile()
result = app.invoke({"topic": "2026 AI trends analysis report"})

# A fully automated content production pipeline
class AIContentPipeline:
    def __init__(self):
        self.researcher = ChatAnthropic(model="claude-4-opus")
        self.writer = ChatOpenAI(model="gpt-4.5-turbo")
        self.image_gen = DALLE4()
        self.video_gen = Sora2()

        # Step 1: In-depth research (Claude)
        research = await self.researcher.ainvoke(f"""
            Research and synthesize the latest trends, statistics, and
            expert opinions about '{topic}' from across the web.
        """)

        # Step 2: Write the content (ChatGPT)

            Based on the research below, write an SEO-optimized blog post:
            {research.content}

            Requirements:
            - At least 2,000 words
            - Propose five compelling title options
            - Optimized keyword density
            - A clear call to action
        """)

        # Step 3: Image generation (DALL-E 4)

            script=article.summary,
            style="professional",
            duration=60,
            aspect_ratio="9:16"
        )

        return {
            "article": article,
            "images": images,
            "video": video,

// AI-powered personalized marketing automation
import { OpenAI } from "openai";
import { Anthropic } from "@anthropic-ai/sdk";

class AIMarketingAutomation {
  constructor() {
    this.segmenter = new OpenAI({ model: "gpt-4.5-turbo" });
    this.copywriter = new Anthropic({ model: "claude-4-sonnet" });
  }

  async createPersonalizedCampaign(customerData) {
    // 1. AI-based automatic customer segmentation
    const segments = await this.segmenter.chat.completions.create({
      messages: [{
        role: "user",
        content: `Analyze the customer data and create the best segments:
        ${JSON.stringify(customerData)}`
      }],
      tools: [{
        type: "function",
        function: {
          name: "analyze_customer_behavior",
          description: "Analyze customer behavior patterns and build segments"
        }
      }]
    });

    // 2. Generate tailored copy per segment (Claude)
    const campaigns = await Promise.all(
      segments.map(segment =>
        this.copywriter.messages.create({
          model: "claude-4-sonnet",
          messages: [{
            role: "user",
            content: `Generate email, social, and ad copy for the following segment:
            Segment: ${segment.name}
            Characteristics: ${segment.characteristics}
            Purchase pattern: ${segment.purchase_history}`
          }]
        })
      )
    );

    // 3. Automatically set up an A/B test
    return this.setupABTest(campaigns);
  }

  async optimizeCampaign(campaignId, performanceData) {
    // Real-time performance analysis and optimization
    const optimization = await this.segmenter.chat.completions.create({
      messages: [{
        role: "system",
        content: "You are a marketing optimization expert."
      }, {
        role: "user",
        content: `Analyze the campaign performance and suggest improvements:
        CTR: ${performanceData.ctr}
        Conversion: ${performanceData.conversion}
        Cost per Acquisition: ${performanceData.cpa}`
      }]
    });

    // Apply optimization automatically
    return this.applyOptimization(campaignId, optimization);
  }
}
