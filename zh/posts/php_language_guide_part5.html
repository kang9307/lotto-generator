PHP 面向对象编程

PHP 从 PHP 5 开始正式支持面向对象编程（OOP），并在后续版本中持续改进相关功能。

类与对象

类的定义与对象的创建

<?php
// 类定义
class User {
    // 属性
    public $name;
    public $email;
    private $password;

    // 构造函数
    public function __construct($name, $email, $password) {
        $this->name = $name;
        $this->email = $email;
        $this->password = $password;
    }

    // 方法
    public function login() {
        return "用户 {$this->email} 已登录。";
    }

    public function getInfo() {
        return "姓名: {$this->name}, 邮箱: {$this->email}";
    }

    // 修改密码方法（访问 private 属性）
    public function changePassword($newPassword) {
        $this->password = $newPassword;
        return "密码已更改。";
    }
}

// 创建对象
$user1 = new User("张三", "zhangsan@example.com", "secret123");

// 调用对象方法
echo $user1->login();  // 用户 zhangsan@example.com 已登录。
echo $user1->getInfo();  // 姓名: 张三, 邮箱: zhangsan@example.com

// 访问公开属性
echo $user1->name;  // 张三
$user1->name = "张四";
echo $user1->name;  // 张四

// 尝试访问私有属性
// echo $user1->password;  // 错误: 无法访问私有属性
?>

访问修饰符

<?php
class Product {
    public $name;        // 可在任何地方访问
    protected $price;    // 仅在本类及其子类中可访问
    private $discount;   // 仅在本类内部可访问

    public function __construct($name, $price, $discount = 0) {
        $this->name = $name;
        $this->price = $price;
        $this->discount = $discount;
    }

    public function getFinalPrice() {
        return $this->price * (1 - $this->discount / 100);
    }

    protected function calculateTax($rate = 10) {
        return $this->getFinalPrice() * ($rate / 100);
    }

    private function applySpecialDiscount() {
        $this->discount += 5;
    }

    public function applyHolidayDiscount() {
        $this->applySpecialDiscount();  // 类自身可以调用私有方法
        return "折扣已应用。";
    }
}

$product = new Product("智能手机", 1000000, 10);
echo $product->name;  // 智能手机
echo $product->getFinalPrice();  // 900000

// 错误: 无法访问 protected 属性
// echo $product->price;

// 错误: 无法访问 protected 方法
// $product->calculateTax();

// 错误: 无法访问 private 方法
// $product->applySpecialDiscount();

// 通过公开方法间接调用私有方法
echo $product->applyHolidayDiscount();  // 折扣已应用。
?>

继承

<?php
// 父类
class Vehicle {
    protected $brand;
    protected $color;

    public function __construct($brand, $color) {
        $this->brand = $brand;
        $this->color = $color;
    }

    public function getInfo() {
        return "品牌: {$this->brand}, 颜色: {$this->color}";
    }

    protected function startEngine() {
        return "引擎启动！";
    }
}

// 子类
class Car extends Vehicle {
    private $doors;

    public function __construct($brand, $color, $doors) {
        parent::__construct($brand, $color);  // 调用父类构造函数
        $this->doors = $doors;
    }

    public function getInfo() {
        // 重写并扩展父类方法
        return parent::getInfo() . ", 车门数: {$this->doors}";
    }

    public function drive() {
        $engineStatus = $this->startEngine();  // 调用继承自父类的 protected 方法
        return $engineStatus . " 汽车正在行驶中。";
    }
}

$car = new Car("现代", "黑色", 4);
echo $car->getInfo();  // 品牌: 现代, 颜色: 黑色, 车门数: 4
echo $car->drive();    // 引擎启动！ 汽车正在行驶中。

// 使用 instanceof 运算符检查对象类型
var_dump($car instanceof Car);      // bool(true)
var_dump($car instanceof Vehicle);  // bool(true)
?>

静态成员

<?php
class Database {
    // 静态属性
    private static $connection = null;
    private static $connectionCount = 0;

    // 静态方法
    public static function connect($host, $user, $password) {
        if (self::$connection === null) {
            // 这里实际上应当建立数据库连接
            self::$connection = "DB connected to $host as $user";
            self::$connectionCount++;
        }

        return self::$connection;
    }

    public static function getConnectionCount() {
        return self::$connectionCount;
    }

    // 实例方法
    public function query($sql) {
        if (self::$connection === null) {
            throw new Exception("必须先建立连接。");
        }

        return "Query executed: $sql";
    }
}

// 调用静态方法
$dbConnection = Database::connect("localhost", "root", "password");
echo $dbConnection;  // DB connected to localhost as root
echo Database::getConnectionCount();  // 1

// 使用实例方法需要先创建实例
$db = new Database();
echo $db->query("SELECT * FROM users");  // Query executed: SELECT * FROM users
?>

抽象类

<?php
// 抽象类
abstract class Shape {
    protected $color;

    public function __construct($color) {
        $this->color = $color;
    }

    // 普通方法
    public function getColor() {
        return $this->color;
    }

    // 抽象方法（子类必须实现）
    abstract public function getArea();
    abstract public function getPerimeter();
}

// 具体类（实现抽象类）
class Circle extends Shape {
    private $radius;

    public function __construct($color, $radius) {
        parent::__construct($color);
        $this->radius = $radius;
    }

    public function getArea() {
        return pi() * $this->radius * $this->radius;
    }

    public function getPerimeter() {
        return 2 * pi() * $this->radius;
    }
}

class Rectangle extends Shape {
    private $width;
    private $height;

    public function __construct($color, $width, $height) {
        parent::__construct($color);
        $this->width = $width;
        $this->height = $height;
    }

    public function getArea() {
        return $this->width * $this->height;
    }

    public function getPerimeter() {
        return 2 * ($this->width + $this->height);
    }
}

// 抽象类不能直接实例化
// $shape = new Shape("红色");  // 错误

$circle = new Circle("蓝色", 5);
echo $circle->getColor();   // 蓝色
echo $circle->getArea();    // 约 78.54
echo $circle->getPerimeter();  // 约 31.42

$rectangle = new Rectangle("绿色", 4, 6);
echo $rectangle->getColor();  // 绿色
echo $rectangle->getArea();   // 24
echo $rectangle->getPerimeter();  // 20
?>

接口

<?php
// 接口定义
interface Drawable {
    public function draw();
}

interface Resizable {
    public function resize($scale);
}

// 实现多个接口
class Square implements Drawable, Resizable {
    private $size;

    public function __construct($size) {
        $this->size = $size;
    }

    public function draw() {
        return "绘制正方形。尺寸: {$this->size}";
    }

    public function resize($scale) {
        $this->size *= $scale;
        return "尺寸已调整为 {$this->size}。";
    }
}

// 结合继承与接口
abstract class UIElement {
    protected $id;

    public function __construct($id) {
        $this->id = $id;
    }

    abstract public function render();
}

class Button extends UIElement implements Drawable, Resizable {
    private $width;
    private $height;

    public function __construct($id, $width, $height) {
        parent::__construct($id);
        $this->width = $width;
        $this->height = $height;
    }

    public function render() {
        return "<button id='{$this->id}'>按钮</button>";
    }

    public function draw() {
        return "绘制按钮。尺寸: {$this->width}x{$this->height}";
    }

    public function resize($scale) {
        $this->width *= $scale;
        $this->height *= $scale;
        return "按钮尺寸已调整为 {$this->width}x{$this->height}。";
    }
}

// 使用接口
$square = new Square(10);
echo $square->draw();    // 绘制正方形。尺寸: 10
echo $square->resize(2); // 尺寸已调整为 20。

$button = new Button("submit-btn", 100, 50);
echo $button->render();  // <button id='submit-btn'>按钮</button>
echo $button->draw();    // 绘制按钮。尺寸: 100x50
echo $button->resize(1.5); // 按钮尺寸已调整为 150x75。

// 在类型提示中使用接口
function drawElements(array $elements) {
    foreach ($elements as $element) {
        if ($element instanceof Drawable) {
            echo $element->draw() . "<br>";
        }
    }
}

drawElements([$square, $button]);
?>

特性 Trait (PHP 5.4+)

<?php
// 定义 Trait
trait Logger {
    private $logCount = 0;

    public function log($message) {
        $this->logCount++;
        return date('Y-m-d H:i:s') . " - $message (日志 #{$this->logCount})";
    }

    public function getLogCount() {
        return $this->logCount;
    }
}

trait FileSystem {
    public function readFile($path) {
        return "读取文件: $path";
    }

    public function writeFile($path, $content) {
        return "写入文件: $path, 内容: $content";
    }
}

// 在类中使用 Trait
class User {
    use Logger;

    private $name;

    public function __construct($name) {
        $this->name = $name;
        $this->log("用户 {$this->name} 已创建");
    }
}

class FileManager {
    // 使用多个 Trait
    use Logger, FileSystem;

    public function copyFile($source, $destination) {
        $this->log("复制文件: $source -> $destination");
        $content = $this->readFile($source);
        $this->writeFile($destination, $content);
        return "文件已复制。";
    }
}

$user = new User("张三");
echo $user->log("尝试登录");  // 2025-06-05 12:34:56 - 尝试登录 (日志 #2)
echo $user->getLogCount();  // 2

$fileManager = new FileManager();
echo $fileManager->log("文件管理器初始化");  // 2025-06-05 12:34:56 - 文件管理器初始化 (日志 #1)
echo $fileManager->copyFile("/path/to/source.txt", "/path/to/destination.txt");
?>

方法链式调用

<?php
class Query {
    private $table;
    private $where = [];
    private $orderBy;
    private $limit;

    public function from($table) {
        $this->table = $table;
        return $this;  // 返回 $this 以实现链式调用
    }

    public function where($column, $operator, $value) {
        $this->where[] = "$column $operator '$value'";
        return $this;
    }

    public function orderBy($column, $direction = 'ASC') {
        $this->orderBy = "$column $direction";
        return $this;
    }

    public function limit($limit) {
        $this->limit = $limit;
        return $this;
    }

    public function build() {
        $sql = "SELECT * FROM {$this->table}";

        if (!empty($this->where)) {
            $sql .= " WHERE " . implode(" AND ", $this->where);
        }

        if ($this->orderBy) {
            $sql .= " ORDER BY {$this->orderBy}";
        }

        if ($this->limit) {
            $sql .= " LIMIT {$this->limit}";
        }

        return $sql;
    }
}

// 使用方法链式调用
$query = new Query();
$sql = $query->from('users')
             ->where('age', '>', 18)
             ->where('status', '=', 'active')
             ->orderBy('created_at', 'DESC')
             ->limit(10)
             ->build();

echo $sql;
// 输出: SELECT * FROM users WHERE age > '18' AND status = 'active' ORDER BY created_at DESC LIMIT 10
?>

魔术方法

<?php
class Person {
    private $data = [];

    // 魔术方法: 访问未定义的属性时触发
    public function __get($name) {
        if (array_key_exists($name, $this->data)) {
            return $this->data[$name];
        }
        return null;
    }

    // 魔术方法: 给未定义的属性赋值时触发
    public function __set($name, $value) {
        $this->data[$name] = $value;
    }

    // 魔术方法: 调用 isset() 时触发
    public function __isset($name) {
        return isset($this->data[$name]);
    }

    // 魔术方法: 调用 unset() 时触发
    public function __unset($name) {
        unset($this->data[$name]);
    }

    // 魔术方法: 对象被转换为字符串时触发
    public function __toString() {
        return "Person 对象: " . json_encode($this->data);
    }

    // 魔术方法: 调用未定义的方法时触发
    public function __call($name, $arguments) {
        $prefix = substr($name, 0, 3);
        if ($prefix === 'get') {
            $property = lcfirst(substr($name, 3));
            return $this->__get($property);
        } elseif ($prefix === 'set') {
            $property = lcfirst(substr($name, 3));
            $this->__set($property, $arguments[0]);
            return $this;
        }
        throw new Exception("方法 '$name' 不存在。");
    }

    // 魔术方法: 调用未定义的静态方法时触发
    public static function __callStatic($name, $arguments) {
        return "静态方法 '$name' 已被调用";
    }

    // 魔术方法: 对象被克隆时触发
    public function __clone() {
        // 执行深拷贝逻辑
        $this->data = array_map(function($item) {
            return is_object($item) ? clone $item : $item;
        }, $this->data);
    }

    // 魔术方法: 调用 serialize() 时触发
    public function __sleep() {
        // 返回需要序列化的属性数组
        return array_keys($this->data);
    }

    // 魔术方法: 调用 unserialize() 时触发
    public function __wakeup() {
        // 反序列化后的初始化工作
    }
}

$person = new Person();
$person->name = "张三";  // 触发 __set()
$person->age = 25;        // 触发 __set()

echo $person->name;       // 触发 __get(): 张三
echo isset($person->name); // 触发 __isset(): true

echo $person->getName();   // 触发 __call(): 张三
$person->setEmail("zhangsan@example.com"); // 触发 __call()

echo $person;             // 触发 __toString(): Person 对象: {"name":"张三","age":25,"email":"zhangsan@example.com"}

unset($person->age);      // 触发 __unset()

echo Person::sayHello();   // 触发 __callStatic(): 静态方法 'sayHello' 已被调用
?>
