PHPの関数と配列

関数

関数とは、特定の処理を行うコードブロックのことで、再利用性とモジュール化のために使われます。

関数の定義と呼び出し

<?php
// 基本的な関数定義
function sayHello() {
    echo "こんにちは！";
}

// 関数の呼び出し
sayHello();  // 出力: こんにちは！

// 引数を持つ関数
function greet($name) {
    echo "こんにちは、$nameさん！";
}

greet("山田");  // 出力: こんにちは、山田さん！

// デフォルト値付き引数
function greetWithTitle($name, $title = "さん") {
    echo "こんにちは、$name$title！";
}

greetWithTitle("山田");  // 出力: こんにちは、山田さん！
greetWithTitle("鈴木", "教授");  // 出力: こんにちは、鈴木教授！

// 戻り値を持つ関数
function add($a, $b) {
    return $a + $b;
}

$result = add(5, 3);  // $result = 8
echo $result;
?>

関数の引数

<?php
// 基本的な引数
function showProfile($name, $age, $city) {
    echo "$nameさんは $age歳で、$cityに住んでいます。";
}

showProfile("山田", 25, "東京");

// デフォルト値付き引数
function setConfig($host, $user = "admin", $port = 3306) {
    echo "ホスト: $host, ユーザー: $user, ポート: $port";
}

setConfig("localhost");  // ホスト: localhost, ユーザー: admin, ポート: 3306
setConfig("example.com", "user1");  // ホスト: example.com, ユーザー: user1, ポート: 3306
setConfig("example.org", "user2", 8080);  // ホスト: example.org, ユーザー: user2, ポート: 8080

// 可変長引数 (PHP 5.6+)
function sum(...$numbers) {
    $total = 0;
    foreach ($numbers as $number) {
        $total += $number;
    }
    return $total;
}

echo sum(1, 2, 3, 4, 5);  // 15

// 配列の展開演算子 (PHP 5.6+)
$values = [1, 2, 3];
echo sum(...$values);  // 6
?>

参照渡し

<?php
// 値渡し (デフォルト)
function incrementValue($value) {
    $value++;
    echo "関数内部: $value<br>";
}

$a = 5;
incrementValue($a);  // 関数内部: 6
echo "関数外部: $a<br>";  // 関数外部: 5 (値は変わらない)

// 参照渡し
function incrementReference(&$value) {
    $value++;
    echo "関数内部: $value<br>";
}

$b = 5;
incrementReference($b);  // 関数内部: 6
echo "関数外部: $b<br>";  // 関数外部: 6 (値が変わる)
?>

型宣言 (PHP 7+)

<?php
// 引数の型宣言
function multiply(int $a, int $b) {
    return $a * $b;
}

echo multiply(5, 3);  // 15
// echo multiply("5", "3");  // PHP 7以前: 15, PHP 7以降: 15 (自動型変換)
// PHP 7のstrictモードではTypeErrorが発生

// 戻り値の型宣言 (PHP 7+)
function divide(float $a, float $b): float {
    return $a / $b;
}

echo divide(10, 3);  // 3.3333333333333

// ユニオン型 (PHP 8+)
function processId(int|string $id): void {
    echo "処理中のID: $id";
}

processId(123);  // 処理中のID: 123
processId("ABC123");  // 処理中のID: ABC123

// ナラブル型 (PHP 7.1+)
function findUser(?string $username): ?array {
    if ($username === null) {
        return null;
    }
    // ユーザー検索ロジック...
    return ["name" => $username, "active" => true];
}

var_dump(findUser(null));  // NULL
var_dump(findUser("admin"));  // ["name" => "admin", "active" => true]
?>

無名関数とクロージャ

<?php
// 無名関数
$greet = function($name) {
    return "こんにちは、$nameさん！";
};

echo $greet("山田");  // こんにちは、山田さん！

// クロージャ (外部変数の利用)
$greeting = "こんにちは";
$greet = function($name) use ($greeting) {
    return "$greeting、$nameさん！";
};

echo $greet("山田");  // こんにちは、山田さん！

// 参照による外部変数の利用
$counter = 0;
$increment = function() use (&$counter) {
    $counter++;
    return $counter;
};

echo $increment();  // 1
echo $increment();  // 2
echo $counter;      // 2

// コールバック関数
$numbers = [1, 2, 3, 4, 5];
$doubled = array_map(function($n) {
    return $n * 2;
}, $numbers);
print_r($doubled);  // [2, 4, 6, 8, 10]
?>

アロー関数 (PHP 7.4+)

<?php
// 従来の無名関数
$add = function($a, $b) {
    return $a + $b;
};

// アロー関数
$add = fn($a, $b) => $a + $b;

echo $add(5, 3);  // 8

// 外部変数の自動キャプチャ
$multiplier = 2;
$multiply = fn($n) => $n * $multiplier;

echo $multiply(5);  // 10

// ネストされたアロー関数
$addAndMultiply = fn($a, $b) => fn($c) => ($a + $b) * $c;

$step1 = $addAndMultiply(2, 3);  // fn($c) => (2 + 3) * $c
echo $step1(4);  // 20 ((2 + 3) * 4)
?>

再帰関数

<?php
// 階乗の計算
function factorial($n) {
    if ($n <= 1) {
        return 1;
    } else {
        return $n * factorial($n - 1);
    }
}

echo factorial(5);  // 120 (5 * 4 * 3 * 2 * 1)

// フィボナッチ数列
function fibonacci($n) {
    if ($n <= 1) {
        return $n;
    } else {
        return fibonacci($n - 1) + fibonacci($n - 2);
    }
}

echo fibonacci(10);  // 55

// ディレクトリ走査
function listFiles($dir) {
    $files = [];

    if (is_dir($dir)) {
        $items = scandir($dir);

        foreach ($items as $item) {
            if ($item != "." && $item != "..") {
                $path = $dir . "/" . $item;

                if (is_dir($path)) {
                    $files = array_merge($files, listFiles($path));
                } else {
                    $files[] = $path;
                }
            }
        }
    }

    return $files;
}

// 注意: 深い再帰はスタックオーバーフローを引き起こす可能性があります
?>

変数のスコープと静的変数

<?php
$globalVar = "グローバル変数";

function testScope() {
    $localVar = "ローカル変数";

    echo $localVar;  // ローカル変数
    // echo $globalVar;  // エラー: 未定義の変数

    global $globalVar;  // グローバル変数へのアクセス
    echo $globalVar;    // グローバル変数

    // 静的変数
    static $counter = 0;
    $counter++;
    echo "カウンター: $counter";
}

testScope();  // ローカル変数、グローバル変数、カウンター: 1
testScope();  // ローカル変数、グローバル変数、カウンター: 2

// スーパーグローバル変数は関数内で常にアクセス可能
function showGlobals() {
    echo $_SERVER['PHP_SELF'];  // スーパーグローバル変数はglobalキーワードなしでアクセス可能
}
?>

配列

配列はPHPで最も便利な複合データ型の一つで、複数の値を一つの変数に格納できます。

配列の生成とアクセス

<?php
// インデックス配列の生成
$fruits = array("りんご", "バナナ", "オレンジ");
$colors = ["赤", "緑", "青"];  // PHP 5.4+の短縮構文

// 配列要素へのアクセス
echo $fruits[0];  // りんご
echo $colors[1];  // 緑

// 配列要素の変更
$fruits[1] = "キウイ";
echo $fruits[1];  // キウイ

// 配列要素の追加
$fruits[] = "マンゴー";  // 配列の末尾に追加
print_r($fruits);  // Array ([0] => りんご [1] => キウイ [2] => オレンジ [3] => マンゴー)

// 連想配列の生成
$person = array(
    "name" => "山田",
    "age" => 25,
    "city" => "東京"
);

$car = [
    "brand" => "トヨタ",
    "model" => "カムリ",
    "year" => 2023
];

// 連想配列へのアクセス
echo $person["name"];  // 山田
echo $car["model"];    // カムリ

// 連想配列の要素変更
$person["age"] = 26;

// 連想配列の要素追加
$person["email"] = "yamada@example.com";
?>

$matrix = [

配列の反復処理

<?php
$fruits = ["りんご", "バナナ", "オレンジ", "キウイ"];

// for文
for ($i = 0; $i < count($fruits); $i++) {
    echo $fruits[$i] . "<br>";
}

// foreach文 (値のみ)
foreach ($fruits as $fruit) {
    echo $fruit . "<br>";
}

// foreach文 (インデックスと値)
foreach ($fruits as $index => $fruit) {
    echo "$index: $fruit<br>";
}

// 連想配列の反復処理
$person = [
    "name" => "山田",
    "age" => 25,
    "city" => "東京"
];

foreach ($person as $key => $value) {
    echo "$key: $value<br>";
}

// 多次元配列の反復処理
$users = [
    ["name" => "山田", "age" => 25],
    ["name" => "鈴木", "age" => 30],
    ["name" => "佐藤", "age" => 28]
];

foreach ($users as $user) {
    echo "名前: " . $user["name"] . ", 年齢: " . $user["age"] . "<br>";
}

// ネストされたforeach
foreach ($users as $index => $user) {
    echo "ユーザー " . ($index + 1) . ":<br>";
    foreach ($user as $key => $value) {
        echo "  $key: $value<br>";
    }
}
?>

配列関数

<?php
$numbers = [5, 3, 8, 1, 2];
$fruits = ["りんご", "バナナ", "オレンジ"];

// 配列の情報
echo count($numbers);           // 5 (配列の長さ)
echo sizeof($numbers);          // 5 (countと同じ)
var_dump(empty($numbers));      // bool(false) (配列が空かどうか確認)
var_dump(isset($numbers[1]));   // bool(true) (要素が存在するか確認)
var_dump(in_array(3, $numbers)); // bool(true) (値が配列にあるか確認)
echo array_search("バナナ", $fruits); // 1 (値のインデックス/キーを検索)

// 配列の操作
$combined = array_merge($numbers, $fruits);  // 配列の結合
$sliced = array_slice($numbers, 1, 3);       // 部分配列 [3, 8, 1]
array_push($fruits, "キウイ", "マンゴー");   // 末尾に要素追加
$lastFruit = array_pop($fruits);             // 末尾の要素を取り除いて返す ("マンゴー")
array_unshift($fruits, "スイカ");            // 先頭に要素追加
$firstFruit = array_shift($fruits);          // 先頭の要素を取り除いて返す ("スイカ")

// 配列の変換
$doubled = array_map(function($n) {          // 各要素に関数を適用
    return $n * 2;
}, $numbers);

$filtered = array_filter($numbers, function($n) {  // 条件に合う要素のみフィルタ
    return $n > 3;
});

$sum = array_reduce($numbers, function($carry, $n) {  // 配列を単一の値に畳み込む
    return $carry + $n;
}, 0);

// ソート
sort($numbers);                // 値でソート (インデックスを再設定)
rsort($numbers);               // 値で降順ソート
asort($person);                // 値でソート (キーを保持)
arsort($person);               // 値で降順ソート (キーを保持)
ksort($person);                // キーでソート
krsort($person);               // キーで降順ソート

// ユーザー定義ソート
usort($users, function($a, $b) {  // 値でユーザー定義ソート
    return $a["age"] <=> $b["age"];  // 年齢でソート
});

// 配列のキーと値
$keys = array_keys($person);    // キーのみ取得
$values = array_values($person); // 値のみ取得
$flipped = array_flip($person);  // キーと値を入れ替え

// 配列の初期化
$filled = array_fill(0, 5, "値");  // 0から始めて5個の要素を"値"で埋める
$range = range(1, 10, 2);  // 1から10まで2ずつ増加する配列 [1, 3, 5, 7, 9]

// 配列の交差と差集合
$array1 = [1, 2, 3, 4, 5];
$array2 = [3, 4, 5, 6, 7];
$intersection = array_intersect($array1, $array2);  // 積集合 [3, 4, 5]
$difference = array_diff($array1, $array2);        // 差集合 [1, 2]
?>

配列の分解 (PHP 7.1+)

<?php
// リスト分解
$info = ["山田", 25, "東京"];
[$name, $age, $city] = $info;

echo $name;  // 山田
echo $age;   // 25
echo $city;  // 東京

// 一部の要素のみ分解
[, $age,] = $info;  // 2番目の要素だけ取得
echo $age;  // 25

// キー付き配列の分解 (PHP 7.1+)
$data = ["name" => "山田", "age" => 25, "city" => "東京"];
["name" => $n, "age" => $a] = $data;

echo $n;  // 山田
echo $a;  // 25

// ネストされた配列の分解
$user = [
    "山田",
    ["東京", "渋谷区"],
    ["PHP", "JavaScript", "Python"]
];

[$username, [$city, $district], $languages] = $user;
echo $username;  // 山田
echo $city;      // 東京
echo $district;  // 渋谷区
echo $languages[0];  // PHP
?>

配列スプレッド (PHP 7.4+)

<?php
$fruits1 = ["りんご", "バナナ"];
$fruits2 = ["オレンジ", "キウイ"];

// PHP 7.4+のスプレッド演算子
$allFruits = [...$fruits1, ...$fruits2];
// ["りんご", "バナナ", "オレンジ", "キウイ"]

// 間に別の要素を追加
$combined = [...$fruits1, "マンゴー", ...$fruits2];
// ["りんご", "バナナ", "マンゴー", "オレンジ", "キウイ"]

// 連想配列のスプレッド
$defaults = ["host" => "localhost", "port" => 3306, "user" => "root"];
$custom = ["user" => "admin", "password" => "secret"];

$config = [...$defaults, ...$custom];
// ["host" => "localhost", "port" => 3306, "user" => "admin", "password" => "secret"]
// 同じキーは後の値で上書きされる
?>
