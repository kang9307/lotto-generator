PHPのオブジェクト指向プログラミング

PHPはPHP 5から本格的にオブジェクト指向プログラミング（OOP）をサポートするようになり、その後のバージョンでも関連機能が継続的に改善されてきました。

クラスとオブジェクト

クラス定義とオブジェクト生成

<?php
// クラス定義
class User {
    // プロパティ
    public $name;
    public $email;
    private $password;

    // コンストラクタ
    public function __construct($name, $email, $password) {
        $this->name = $name;
        $this->email = $email;
        $this->password = $password;
    }

    // メソッド
    public function login() {
        return "ユーザー {$this->email} がログインしました。";
    }

    public function getInfo() {
        return "名前: {$this->name}, メール: {$this->email}";
    }

    // パスワード変更メソッド (privateプロパティへのアクセス)
    public function changePassword($newPassword) {
        $this->password = $newPassword;
        return "パスワードが変更されました。";
    }
}

// オブジェクト生成
$user1 = new User("山田", "yamada@example.com", "secret123");

// オブジェクトのメソッド呼び出し
echo $user1->login();  // ユーザー yamada@example.com がログインしました。
echo $user1->getInfo();  // 名前: 山田, メール: yamada@example.com

// publicプロパティへのアクセス
echo $user1->name;  // 山田
$user1->name = "山田花子";
echo $user1->name;  // 山田花子

// privateプロパティへのアクセス試行
// echo $user1->password;  // エラー: privateプロパティにはアクセスできません
?>

アクセス修飾子

<?php
class Product {
    public $name;        // どこからでもアクセス可能
    protected $price;    // このクラスと子クラスからのみアクセス可能
    private $discount;   // このクラス内でのみアクセス可能

    public function __construct($name, $price, $discount = 0) {
        $this->name = $name;
        $this->price = $price;
        $this->discount = $discount;
    }

    public function getFinalPrice() {
        return $this->price * (1 - $this->discount / 100);
    }

    protected function calculateTax($rate = 10) {
        return $this->getFinalPrice() * ($rate / 100);
    }

    private function applySpecialDiscount() {
        $this->discount += 5;
    }

    public function applyHolidayDiscount() {
        $this->applySpecialDiscount();  // 自身のprivateメソッドは呼び出し可能
        return "割引が適用されました。";
    }
}

$product = new Product("スマートフォン", 1000000, 10);
echo $product->name;  // スマートフォン
echo $product->getFinalPrice();  // 900000

// エラー: protectedプロパティにはアクセス不可
// echo $product->price;

// エラー: protectedメソッドにはアクセス不可
// $product->calculateTax();

// エラー: privateメソッドにはアクセス不可
// $product->applySpecialDiscount();

// publicメソッドを通じてprivateメソッドを間接的に呼び出し
echo $product->applyHolidayDiscount();  // 割引が適用されました。
?>

継承

<?php
// 親クラス
class Vehicle {
    protected $brand;
    protected $color;

    public function __construct($brand, $color) {
        $this->brand = $brand;
        $this->color = $color;
    }

    public function getInfo() {
        return "ブランド: {$this->brand}, 色: {$this->color}";
    }

    protected function startEngine() {
        return "エンジン始動！";
    }
}

// 子クラス
class Car extends Vehicle {
    private $doors;

    public function __construct($brand, $color, $doors) {
        parent::__construct($brand, $color);  // 親コンストラクタの呼び出し
        $this->doors = $doors;
    }

    public function getInfo() {
        // 親メソッドのオーバーライドと拡張
        return parent::getInfo() . ", ドア数: {$this->doors}";
    }

    public function drive() {
        $engineStatus = $this->startEngine();  // 継承したprotectedメソッドの呼び出し
        return $engineStatus . " 車が走行中です。";
    }
}

$car = new Car("トヨタ", "黒", 4);
echo $car->getInfo();  // ブランド: トヨタ, 色: 黒, ドア数: 4
echo $car->drive();    // エンジン始動！ 車が走行中です。

// instanceof演算子でオブジェクトの型を確認
var_dump($car instanceof Car);      // bool(true)
var_dump($car instanceof Vehicle);  // bool(true)
?>

静的メンバ

<?php
class Database {
    // 静的プロパティ
    private static $connection = null;
    private static $connectionCount = 0;

    // 静的メソッド
    public static function connect($host, $user, $password) {
        if (self::$connection === null) {
            // 実際にはここでデータベース接続を確立する
            self::$connection = "DB connected to $host as $user";
            self::$connectionCount++;
        }

        return self::$connection;
    }

    public static function getConnectionCount() {
        return self::$connectionCount;
    }

    // 通常メソッド
    public function query($sql) {
        if (self::$connection === null) {
            throw new Exception("先に接続を確立する必要があります。");
        }

        return "Query executed: $sql";
    }
}

// 静的メソッドの呼び出し
$dbConnection = Database::connect("localhost", "root", "password");
echo $dbConnection;  // DB connected to localhost as root
echo Database::getConnectionCount();  // 1

// 通常メソッドを使うにはインスタンスが必要
$db = new Database();
echo $db->query("SELECT * FROM users");  // Query executed: SELECT * FROM users
?>

抽象クラス

<?php
// 抽象クラス
abstract class Shape {
    protected $color;

    public function __construct($color) {
        $this->color = $color;
    }

    // 通常メソッド
    public function getColor() {
        return $this->color;
    }

    // 抽象メソッド (子クラスで必ず実装する必要あり)
    abstract public function getArea();
    abstract public function getPerimeter();
}

// 具象クラス (抽象クラスの実装)
class Circle extends Shape {
    private $radius;

    public function __construct($color, $radius) {
        parent::__construct($color);
        $this->radius = $radius;
    }

    public function getArea() {
        return pi() * $this->radius * $this->radius;
    }

    public function getPerimeter() {
        return 2 * pi() * $this->radius;
    }
}

class Rectangle extends Shape {
    private $width;
    private $height;

    public function __construct($color, $width, $height) {
        parent::__construct($color);
        $this->width = $width;
        $this->height = $height;
    }

    public function getArea() {
        return $this->width * $this->height;
    }

    public function getPerimeter() {
        return 2 * ($this->width + $this->height);
    }
}

// 抽象クラスはインスタンス化できない
// $shape = new Shape("赤");  // エラー

$circle = new Circle("青", 5);
echo $circle->getColor();   // 青
echo $circle->getArea();    // 約 78.54
echo $circle->getPerimeter();  // 約 31.42

$rectangle = new Rectangle("緑", 4, 6);
echo $rectangle->getColor();  // 緑
echo $rectangle->getArea();   // 24
echo $rectangle->getPerimeter();  // 20
?>

インターフェース

<?php
// インターフェース定義
interface Drawable {
    public function draw();
}

interface Resizable {
    public function resize($scale);
}

// 複数のインターフェースを実装
class Square implements Drawable, Resizable {
    private $size;

    public function __construct($size) {
        $this->size = $size;
    }

    public function draw() {
        return "正方形を描画します。サイズ: {$this->size}";
    }

    public function resize($scale) {
        $this->size *= $scale;
        return "サイズが {$this->size} に変更されました。";
    }
}

// 継承とインターフェースの組み合わせ
abstract class UIElement {
    protected $id;

    public function __construct($id) {
        $this->id = $id;
    }

    abstract public function render();
}

class Button extends UIElement implements Drawable, Resizable {
    private $width;
    private $height;

    public function __construct($id, $width, $height) {
        parent::__construct($id);
        $this->width = $width;
        $this->height = $height;
    }

    public function render() {
        return "<button id='{$this->id}'>ボタン</button>";
    }

    public function draw() {
        return "ボタンを描画します。サイズ: {$this->width}x{$this->height}";
    }

    public function resize($scale) {
        $this->width *= $scale;
        $this->height *= $scale;
        return "ボタンサイズが {$this->width}x{$this->height} に変更されました。";
    }
}

// インターフェースの使用
$square = new Square(10);
echo $square->draw();    // 正方形を描画します。サイズ: 10
echo $square->resize(2); // サイズが 20 に変更されました。

$button = new Button("submit-btn", 100, 50);
echo $button->render();  // <button id='submit-btn'>ボタン</button>
echo $button->draw();    // ボタンを描画します。サイズ: 100x50
echo $button->resize(1.5); // ボタンサイズが 150x75 に変更されました。

// 型ヒントにインターフェースを使用
function drawElements(array $elements) {
    foreach ($elements as $element) {
        if ($element instanceof Drawable) {
            echo $element->draw() . "<br>";
        }
    }
}

drawElements([$square, $button]);
?>

トレイト (PHP 5.4+)

<?php
// トレイト定義
trait Logger {
    private $logCount = 0;

    public function log($message) {
        $this->logCount++;
        return date('Y-m-d H:i:s') . " - $message (ログ #{$this->logCount})";
    }

    public function getLogCount() {
        return $this->logCount;
    }
}

trait FileSystem {
    public function readFile($path) {
        return "ファイル読み込み: $path";
    }

    public function writeFile($path, $content) {
        return "ファイル書き込み: $path, 内容: $content";
    }
}

// クラスでトレイトを使用
class User {
    use Logger;

    private $name;

    public function __construct($name) {
        $this->name = $name;
        $this->log("ユーザー {$this->name} を作成");
    }
}

class FileManager {
    // 複数のトレイトを使用
    use Logger, FileSystem;

    public function copyFile($source, $destination) {
        $this->log("ファイルコピー: $source -> $destination");
        $content = $this->readFile($source);
        $this->writeFile($destination, $content);
        return "ファイルがコピーされました。";
    }
}

$user = new User("山田");
echo $user->log("ログイン試行");  // 2025-06-05 12:34:56 - ログイン試行 (ログ #2)
echo $user->getLogCount();  // 2

$fileManager = new FileManager();
echo $fileManager->log("ファイルマネージャー初期化");  // 2025-06-05 12:34:56 - ファイルマネージャー初期化 (ログ #1)
echo $fileManager->copyFile("/path/to/source.txt", "/path/to/destination.txt");
?>

メソッドチェーン

<?php
class Query {
    private $table;
    private $where = [];
    private $orderBy;
    private $limit;

    public function from($table) {
        $this->table = $table;
        return $this;  // メソッドチェーンのため $this を返す
    }

    public function where($column, $operator, $value) {
        $this->where[] = "$column $operator '$value'";
        return $this;
    }

    public function orderBy($column, $direction = 'ASC') {
        $this->orderBy = "$column $direction";
        return $this;
    }

    public function limit($limit) {
        $this->limit = $limit;
        return $this;
    }

    public function build() {
        $sql = "SELECT * FROM {$this->table}";

        if (!empty($this->where)) {
            $sql .= " WHERE " . implode(" AND ", $this->where);
        }

        if ($this->orderBy) {
            $sql .= " ORDER BY {$this->orderBy}";
        }

        if ($this->limit) {
            $sql .= " LIMIT {$this->limit}";
        }

        return $sql;
    }
}

// メソッドチェーンの使用
$query = new Query();
$sql = $query->from('users')
             ->where('age', '>', 18)
             ->where('status', '=', 'active')
             ->orderBy('created_at', 'DESC')
             ->limit(10)
             ->build();

echo $sql;
// 出力: SELECT * FROM users WHERE age > '18' AND status = 'active' ORDER BY created_at DESC LIMIT 10
?>

マジックメソッド

<?php
class Person {
    private $data = [];

    // マジックメソッド: 未定義のプロパティにアクセスしたときに呼ばれる
    public function __get($name) {
        if (array_key_exists($name, $this->data)) {
            return $this->data[$name];
        }
        return null;
    }

    // マジックメソッド: 未定義のプロパティへ代入したときに呼ばれる
    public function __set($name, $value) {
        $this->data[$name] = $value;
    }

    // マジックメソッド: isset() 呼び出し時に呼ばれる
    public function __isset($name) {
        return isset($this->data[$name]);
    }

    // マジックメソッド: unset() 呼び出し時に呼ばれる
    public function __unset($name) {
        unset($this->data[$name]);
    }

    // マジックメソッド: オブジェクトを文字列に変換するときに呼ばれる
    public function __toString() {
        return "Personオブジェクト: " . json_encode($this->data);
    }

    // マジックメソッド: 未定義のメソッド呼び出し時に呼ばれる
    public function __call($name, $arguments) {
        $prefix = substr($name, 0, 3);
        if ($prefix === 'get') {
            $property = lcfirst(substr($name, 3));
            return $this->__get($property);
        } elseif ($prefix === 'set') {
            $property = lcfirst(substr($name, 3));
            $this->__set($property, $arguments[0]);
            return $this;
        }
        throw new Exception("メソッド '$name' は存在しません。");
    }

    // マジックメソッド: 未定義の静的メソッド呼び出し時に呼ばれる
    public static function __callStatic($name, $arguments) {
        return "静的メソッド '$name' が呼び出されました";
    }

    // マジックメソッド: オブジェクトがクローンされるときに呼ばれる
    public function __clone() {
        // ディープコピーのためのロジック
        $this->data = array_map(function($item) {
            return is_object($item) ? clone $item : $item;
        }, $this->data);
    }

    // マジックメソッド: serialize() 呼び出し時に呼ばれる
    public function __sleep() {
        // シリアライズするプロパティの配列を返す
        return array_keys($this->data);
    }

    // マジックメソッド: unserialize() 呼び出し時に呼ばれる
    public function __wakeup() {
        // デシリアライズ後の初期化処理
    }
}

$person = new Person();
$person->name = "山田";  // __set() 呼び出し
$person->age = 25;        // __set() 呼び出し

echo $person->name;       // __get() 呼び出し: 山田
echo isset($person->name); // __isset() 呼び出し: true

echo $person->getName();   // __call() 呼び出し: 山田
$person->setEmail("yamada@example.com"); // __call() 呼び出し

echo $person;             // __toString() 呼び出し: Personオブジェクト: {"name":"山田","age":25,"email":"yamada@example.com"}

unset($person->age);      // __unset() 呼び出し

echo Person::sayHello();   // __callStatic() 呼び出し: 静的メソッド 'sayHello' が呼び出されました
?>
